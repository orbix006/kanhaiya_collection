--  ENTERPRISE E-COMMERCE + ADMIN PANEL  |  SUPABASE (PostgreSQL 15+)
--  v3 = BASELINE MIGRATION (0001). Run once on an EMPTY project.
--
--  What is new in v3 (vs v2):
--   * All review findings fixed (sanitizer, FOUND bug, replayable webhooks,
--     stock overwrite, last admin bypass, deactivation, default privileges,
--     column level exposure, campaign window leak, throttling, PII in audit,
--     storage listing, IST timezone, voided payments, retention, cart,
--     GST inclusive pricing, refund caps, RLS initplan wrapping, indexes).
--   * Razorpay online payment logic (order intent, signature verified
--     confirmation, webhooks, gateway refunds, late payment detection).
--   * Automatic GST style invoice issued the moment an order is confirmed,
--     immutable, gapless numbering per financial year, emailed through an
--     outbox, downloadable by the customer from a private bucket.
--
--  Safe by default: reset switch is FALSE; the script refuses to run again
--  once public.profiles exists. Future changes ship as NEW migration files.
--
--  RAZORPAY FLOW (all secrets live in Edge Function env, never in the DB)
--   1. Client calls place_order(items, address, 'online', ...)
--        -> order (pending), stock reserved, one pending charge intent row.
--   2. Edge fn create-razorpay-order (user JWT):
--        rpc begin_online_payment(order_id)           -> amount, existing rzp order id
--        POST razorpay /v1/orders {amount, currency, receipt=order_number,
--                                  notes:{order_id}}   (skip if already attached)
--        rpc attach_razorpay_order(...) [service_role]
--        return key_id + razorpay order id to the browser.
--   3. Browser opens Razorpay Checkout. On success it receives
--        razorpay_payment_id, razorpay_order_id, razorpay_signature.
--   4. Edge fn verify-razorpay-payment (user JWT): verify
--        HMAC_SHA256(order_id|payment_id, key_secret), then fetch the payment
--        from Razorpay and confirm status captured, then call
--        apply_payment_event(provider 'razorpay', event_id 'client:'||payment_id,
--        'payment.captured', ...) [service_role]. Instant UX; idempotent.
--   5. Edge fn payment-webhook (public, HMAC verified with webhook secret) is the
--        source of truth. Map X-Razorpay-Event-Id -> p_event_id and call
--        apply_payment_event for: payment.captured, order.paid, payment.failed,
--        refund.processed, refund.failed. Amounts are passed in PAISE.
--        Return 200 for status processed, duplicate, ignored AND failed
--        (failed rows are visible in v_payment_events_failed; alert on them).
--   6. Capture success => payments ledger row succeeded, orders.payment_status
--        derived 'paid', orders.confirmed_at set, issue_invoice() runs, an
--        outbox row is queued. Edge fn send-invoices drains the outbox:
--        claim_outbox -> render PDF -> upload to bucket invoices/<user_id>/<order_number>.pdf
--        -> set_invoice_pdf -> email -> mark_outbox.
--   7. Refunds (admin): Edge fn refund (admin JWT verified, then service_role):
--        begin_gateway_refund -> POST razorpay /v1/payments/:id/refund ->
--        finish_gateway_refund. The refund becomes succeeded only when the
--        refund.processed webhook arrives.
--   8. Payment arriving after the order was cancelled or expired is stored in
--        the ledger and shows up in v_orders_needing_refund. No invoice is issued.
--
--  Razorpay dashboard: enable auto capture, subscribe the webhook to the five
--  events above. Never mark an order paid from the client.
-- =====================================================================

    end loop;

    for r in select unnest(array['v_products_admin','v_storefront_products','v_storefront_categories',
        'v_active_campaigns','v_low_stock_products','v_orders_needing_refund','v_payment_events_failed']) as v loop
      execute format('drop view if exists public.%I cascade', r.v);
    end loop;

    for r in select p.oid::regprocedure as sig from pg_proc p
             where p.pronamespace = 'public'::regnamespace
               and not exists (select 1 from pg_depend d where d.objid = p.oid and d.deptype = 'e') loop
      execute 'drop function if exists ' || r.sig || ' cascade';
    end loop;

    for r in select unnest(array['user_role','order_status','payment_status','lead_status','policy_type',
                                 'discount_type','return_status']) as t loop
      execute format('drop type if exists public.%I cascade', r.t);
    end loop;

    for r in select unnest(array['product_sku_seq','product_batch_seq','order_number_seq']) as s loop
      execute format('drop sequence if exists public.%I', r.s);
    end loop;

    for r in select polname from pg_policy
             where polrelid = 'storage.objects'::regclass and polname like 'shop\_%' loop
      execute format('drop policy if exists %I on storage.objects', r.polname);
    end loop;
  elsif to_regclass('public.profiles') is not null then
    raise exception 'Schema already installed. This file is baseline migration 0001; apply changes as new migration files.';
  end if;
end

create or replace function public.set_updated_at()
returns trigger language plpgsql as $$
begin new.updated_at = now(); return new; end $$;

create or replace function public.slugify(txt text)
returns text language sql immutable as $$
  select trim(both '-' from regexp_replace(lower(coalesce(txt, '')), '[^a-z0-9]+', '-', 'g'));
$$;

create or replace function public.escape_like(t text)
returns text language sql immutable as $$
  select replace(replace(replace(coalesce(t, ''), '\', '\\'), '%', '\%'), '_', '\_');
$$;

-- Best effort defence in depth ONLY. Always sanitize at render time (e.g. DOMPurify).
-- Postgres regex: \y is the word boundary (\b would mean backspace).
create or replace function public.sanitize_html(t text)
returns text language sql immutable as $$
  select regexp_replace(regexp_replace(regexp_replace(regexp_replace(coalesce(t, ''),
    '<\s*?(script|style|iframe|object|embed|link|meta|base|form)\y[^>]*>.*?<\s*/\s*\1\s*>', '', 'gi'),
    '<\s*/?\s*(script|style|iframe|object|embed|link|meta|base|form)\y[^>]*>', '', 'gi'),
    '\son\w+\s*=\s*("[^"]*"|''[^'']*''|[^\s>]+)', '', 'gi'),
    'javascript\s*:', '', 'gi');
$$;

-- Atomic fixed window rate limiter (race free: single upsert)
create table public.rate_limits (
  key          text not null,
  window_start timestamptz not null,
  hits         int not null default 0,
  primary key (key, window_start)
);

create or replace function public.rate_limit_hit(p_key text, p_limit int, p_window_seconds int)
returns boolean language plpgsql security definer set search_path = public as $$
declare v_start timestamptz; v_hits int;
begin
  v_start := to_timestamp(floor(extract(epoch from now()) / p_window_seconds) * p_window_seconds);
  insert into public.rate_limits(key, window_start, hits) values (p_key, v_start, 1)
  on conflict (key, window_start) do update set hits = public.rate_limits.hits + 1
  returning hits into v_hits;
  return v_hits <= p_limit;
end $$;

create or replace function public.purge_rate_limits()
returns bigint language plpgsql security definer set search_path = public as $$
declare n bigint;
begin
  delete from public.rate_limits where window_start < now() - interval '1 day';
  get diagnostics n = row_count; return n;
end $$;

  insert into public.profiles (id, email, full_name, phone, avatar_url)

create or replace function public.handle_user_email_change()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  perform set_config('app.profile_admin_op', 'on', true);
  update public.profiles set email = new.email where id = new.id;
  perform set_config('app.profile_admin_op', '', true);
  return new;
end $$;
create trigger on_auth_user_email_changed after update of email on auth.users
  for each row when (old.email is distinct from new.email)
  execute function public.handle_user_email_change();

-- Role, status and email can only change through the audited admin RPCs
-- (they set app.profile_admin_op) or from service_role / SQL editor (auth.uid() is null).
-- This closes the direct UPDATE route that bypassed the "last admin" guard.
create or replace function public.guard_profile_update()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  if auth.uid() is not null and coalesce(current_setting('app.profile_admin_op', true), '') <> 'on' then
    if new.role is distinct from old.role then raise exception 'Not allowed to change role'; end if;
    if new.is_active is distinct from old.is_active then raise exception 'Not allowed to change account status'; end if;
    new.email := old.email;
  end if;
  return new;
end $$;
create trigger trg_guard_profile_update before update on public.profiles
  for each row execute function public.guard_profile_update();

-- Deactivating a profile also bans the auth user (blocks new sign ins and token refresh).
-- Already issued JWTs stay valid until expiry, so the RPCs and RLS also check is_active.
create or replace function public.profiles_sync_auth_ban()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  begin
    update auth.users
       set banned_until = case when new.is_active then null else 'infinity'::timestamptz end
     where id = new.id;
  exception when others then
    raise warning 'Could not sync auth ban for %: %', new.id, sqlerrm;
  end;
  return null;
end $$;
create trigger trg_profiles_ban after update of is_active on public.profiles
  for each row when (old.is_active is distinct from new.is_active)
  execute function public.profiles_sync_auth_ban();

create or replace function public.admin_set_user_role(p_user uuid, p_role public.user_role)
returns public.profiles language plpgsql security definer set search_path = public as $$
declare v public.profiles;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  if p_role = 'user' and (select role from public.profiles where id = p_user) = 'admin'
     and (select count(*) from public.profiles where role = 'admin' and is_active and id <> p_user) = 0 then
    raise exception 'Cannot remove the last active admin';
  end if;
  perform set_config('app.profile_admin_op', 'on', true);
  update public.profiles set role = p_role where id = p_user returning * into v;
  perform set_config('app.profile_admin_op', '', true);
  if v.id is null then raise exception 'User not found'; end if;
  return v;
end $$;

create or replace function public.admin_set_user_active(p_user uuid, p_active boolean)
returns public.profiles language plpgsql security definer set search_path = public as $$
declare v public.profiles;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  if not p_active and p_user = auth.uid() then raise exception 'You cannot deactivate yourself'; end if;
  if not p_active and (select role from public.profiles where id = p_user) = 'admin'
     and (select count(*) from public.profiles where role = 'admin' and is_active and id <> p_user) = 0 then
    raise exception 'Cannot deactivate the last active admin';
  end if;
  perform set_config('app.profile_admin_op', 'on', true);
  update public.profiles set is_active = p_active where id = p_user returning * into v;
  perform set_config('app.profile_admin_op', '', true);
  if v.id is null then raise exception 'User not found'; end if;
  return v;
end $$;

create or replace function public.admin_list_users(
  p_search text default null, p_role public.user_role default null,
  p_page int default 1, p_page_size int default 10)
returns jsonb language plpgsql stable security definer set search_path = public as $$
declare v_total bigint; v_rows jsonb;
        v_size int := least(greatest(p_page_size, 1), 100);
        v_off int := greatest(p_page - 1, 0) * least(greatest(p_page_size, 1), 100);
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  select count(*) into v_total from public.profiles u
   where (p_role is null or u.role = p_role)
     and (coalesce(p_search, '') = '' or (u.email || ' ' || coalesce(u.full_name, '') || ' ' || coalesce(u.phone, ''))
          ilike '%' || public.escape_like(p_search) || '%');
  select coalesce(jsonb_agg(to_jsonb(t)), '[]'::jsonb) into v_rows from (
    select u.id, u.email, u.full_name, u.phone, u.role, u.is_active, u.created_at
      from public.profiles u
     where (p_role is null or u.role = p_role)
       and (coalesce(p_search, '') = '' or (u.email || ' ' || coalesce(u.full_name, '') || ' ' || coalesce(u.phone, ''))
            ilike '%' || public.escape_like(p_search) || '%')
     order by u.created_at desc offset v_off limit v_size) t;
  return jsonb_build_object('total', v_total, 'page', p_page, 'page_size', v_size, 'offset', v_off, 'rows', v_rows);
end $$;

-- Optional JWT claim "user_role" (Dashboard > Authentication > Hooks > Custom Access Token)
create or replace function public.custom_access_token_hook(event jsonb)
returns jsonb language plpgsql stable as $$
declare claims jsonb; v_role public.user_role;
begin
  select role into v_role from public.profiles where id = (event->>'user_id')::uuid;
  claims := coalesce(event->'claims', '{}'::jsonb);
  claims := jsonb_set(claims, '{user_role}', to_jsonb(coalesce(v_role, 'user'::public.user_role)));
  return jsonb_set(event, '{claims}', claims);
end $$;

-- Indian financial year label: April 2026 to March 2027 => '2627'
create or replace function public.fy_label(d date)
returns text language sql immutable as $$
  select case when extract(month from d) >= 4
              then to_char(d, 'YY') || to_char(d + interval '1 year', 'YY')
              else to_char(d - interval '1 year', 'YY') || to_char(d, 'YY') end;
$$;

-- ---------------------------------------------------------------------
-- 7. STORAGE CLEANUP QUEUE
--    Drain with an Edge Function/cron through the Storage API (service_role).
-- ---------------------------------------------------------------------
create table public.storage_cleanup_queue (
  id         bigint generated always as identity primary key,
  bucket     text not null,
  path       text not null,
  created_at timestamptz not null default now(),
  unique (bucket, path)
);

create or replace function public.enqueue_storage_cleanup()
returns trigger language plpgsql security definer set search_path = public as $$
declare v_old text; v_new text;
begin
  v_old := to_jsonb(old)->>tg_argv[1];
  if tg_op = 'UPDATE' then v_new := to_jsonb(new)->>tg_argv[1]; end if;
  if v_old is not null and v_old is distinct from v_new then
    insert into public.storage_cleanup_queue(bucket, path) values (tg_argv[0], v_old) on conflict do nothing;
  end if;
  return null;
end $$;

-- 8. CATEGORIES  (category > subcategory, atomic drag and drop)

create table public.categories (

create index categories_parent_sort_idx on public.categories(parent_id, sort_order);
create unique index categories_unique_name_per_parent
  on public.categories (coalesce(parent_id, '00000000-0000-0000-0000-000000000000'::uuid), lower(name));
create unique index categories_unique_slug_per_parent
  on public.categories (coalesce(parent_id, '00000000-0000-0000-0000-000000000000'::uuid), lower(slug));
create trigger trg_categories_updated before update on public.categories
  for each row execute function public.set_updated_at();
create trigger trg_categories_storage after update of thumbnail_path or delete on public.categories

create or replace function public.category_visible(p_id uuid)
returns boolean language sql stable security definer set search_path = public as $$
  select coalesce((select is_visible from public.categories where id = p_id), false);
$$;

create or replace function public.categories_before_write()
returns trigger language plpgsql as $$
declare v_pp uuid;
begin
  if new.slug is null or new.slug = '' then
    new.slug := public.slugify(new.name) || '-' || substr(gen_random_uuid()::text, 1, 6);
  end if;
  if new.parent_id is not null and coalesce(current_setting('app.bulk_reorder', true), '') <> 'on' then
    select parent_id into v_pp from public.categories where id = new.parent_id;
    if not found then raise exception 'Parent category not found'; end if;
    if v_pp is not null then raise exception 'Only two levels supported: category > subcategory'; end if;
    if tg_op = 'UPDATE' then
      if exists (select 1 from public.categories where parent_id = new.id) then
        raise exception 'A category that has subcategories cannot become a subcategory';
      end if;
      if exists (select 1 from public.products where category_id = new.id) then
        raise exception 'A category that directly holds products cannot become a subcategory';
      end if;
    end if;
  end if;
  return new;
end $$;
create trigger trg_categories_guard before insert or update on public.categories
  for each row execute function public.categories_before_write();

create or replace function public.categories_recalc_visibility()
returns trigger language plpgsql security definer set search_path = public as $$
declare v_prev text := coalesce(current_setting('app.bulk_reorder', true), '');
begin
  perform set_config('app.bulk_reorder', 'on', true);
  update public.categories c
     set is_visible = c.is_active and (c.parent_id is null
           or coalesce((select p.is_active from public.categories p where p.id = c.parent_id), false))
   where c.id = new.id or c.parent_id = new.id;
  perform set_config('app.bulk_reorder', v_prev, true);
  return null;
end $$;
create trigger trg_categories_visibility after insert or update of is_active, parent_id on public.categories
  for each row execute function public.categories_recalc_visibility();

create or replace function public.categories_after_move()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  if new.parent_id is distinct from old.parent_id then
    if new.parent_id is null then
      update public.products set subcategory_id = null where subcategory_id = new.id;
    else
      update public.products set category_id = new.parent_id where subcategory_id = new.id;
    end if;
  end if;
  return null;
end $$;
create trigger trg_categories_move after update of parent_id on public.categories
  for each row execute function public.categories_after_move();

-- Atomic drag and drop. Validates EVERY id/parent first, applies in one statement,
-- then validates the final tree; any failure rolls everything back.
-- p_items: [{"id":"..","parent_id":null|"..","sort_order":0}, ...]
create or replace function public.reorder_categories(p_items jsonb)
returns void language plpgsql security definer set search_path = public as $$
declare v_ids uuid[]; v_n int;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  if p_items is null or jsonb_typeof(p_items) <> 'array' or jsonb_array_length(p_items) = 0 then
    raise exception 'Nothing to reorder';
  end if;
  select array_agg((x->>'id')::uuid) into v_ids from jsonb_array_elements(p_items) x;
  v_n := array_length(v_ids, 1);
  if (select count(distinct u) from unnest(v_ids) u) <> v_n then raise exception 'Duplicate category ids in payload'; end if;
  if (select count(*) from public.categories where id = any (v_ids)) <> v_n then raise exception 'Unknown category id in payload'; end if;
  if exists (select 1 from jsonb_array_elements(p_items) x
              where nullif(x->>'parent_id', '') is not null
                and not exists (select 1 from public.categories c where c.id = (x->>'parent_id')::uuid)) then
    raise exception 'Unknown parent category in payload';
  end if;

  perform set_config('app.bulk_reorder', 'on', true);
  update public.categories c
     set parent_id  = nullif(x->>'parent_id', '')::uuid,
         sort_order = coalesce((x->>'sort_order')::int, c.sort_order)
    from jsonb_array_elements(p_items) x
   where c.id = (x->>'id')::uuid;
  perform set_config('app.bulk_reorder', '', true);

  if exists (select 1 from public.categories c join public.categories p on p.id = c.parent_id where p.parent_id is not null) then
    raise exception 'Only two levels supported: category > subcategory';
  end if;
  if exists (select 1 from public.products pr join public.categories c on c.id = pr.category_id where c.parent_id is not null) then
    raise exception 'A category that holds products cannot become a subcategory';
  end if;
end $$;

create or replace function public.products_before_write()
returns trigger language plpgsql as $$
begin
  if new.slug is null or new.slug = '' then
    new.slug := public.slugify(new.name) || '-' || substr(gen_random_uuid()::text, 1, 6);
  end if;
  if not exists (select 1 from public.categories where id = new.category_id and parent_id is null) then
    raise exception 'Product category must be a top level category';
  end if;
  if new.subcategory_id is not null and not exists (
       select 1 from public.categories where id = new.subcategory_id and parent_id = new.category_id) then
    raise exception 'Subcategory does not belong to the selected category';
  end if;
  if tg_op = 'UPDATE' then new.sku := old.sku; end if;   -- SKU is immutable, batch_no is editable
  return new;
end $$;

returns trigger language plpgsql security definer set search_path = public as $$
begin
  if tg_op = 'INSERT' then
    insert into public.inventory_movements(product_id, delta, stock_after, reason, actor_id)

  return null;
end $$;

create or replace view public.v_products_admin with (security_invoker = true) as
select p.*, c.name as category_name, s.name as subcategory_name,
       (p.stock <= p.low_stock_threshold) as is_low_stock,
       (select i.url from public.product_images i where i.product_id = p.id
         order by i.is_primary desc, i.sort_order limit 1) as primary_image

select * from public.v_products_admin where is_low_stock;

-- Stock is accepted ONLY on insert. Later changes go through admin_adjust_stock
-- (a relative delta), so a stale edit form can never overwrite live stock.
create or replace function public.admin_save_product(p jsonb)
returns jsonb language plpgsql security definer set search_path = public as $$
declare
  v_id uuid := nullif(p->>'id', '')::uuid;
  v_batch text := nullif(trim(p->>'batch_no'), '');
  v_prod public.products; img record; v_prim int; v_removed jsonb := '[]'::jsonb; v_urls text[];
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;

  if v_id is null then
    insert into public.products(name, slug, description, category_id, subcategory_id, batch_no, hsn_code, price, stock,
                                low_stock_threshold, is_active, created_by)
    values (trim(p->>'name'), coalesce(nullif(p->>'slug', ''), ''), p->>'description',
            (p->>'category_id')::uuid, nullif(p->>'subcategory_id', '')::uuid,
            coalesce(v_batch, 'B' || to_char(now(), 'YYYYMMDD') || '-' || lpad(nextval('public.product_batch_seq')::text, 4, '0')),
            nullif(trim(p->>'hsn_code'), ''),
            (p->>'price')::numeric, coalesce((p->>'stock')::int, 0),
            coalesce((p->>'low_stock_threshold')::int, (select low_stock_default from public.store_settings limit 1)),
            coalesce((p->>'is_active')::boolean, true), auth.uid())
    returning * into v_prod;
  else
    update public.products set
      name           = coalesce(nullif(trim(p->>'name'), ''), name),
      description    = case when p ? 'description' then p->>'description' else description end,
      category_id    = coalesce((p->>'category_id')::uuid, category_id),
      subcategory_id = case when p ? 'subcategory_id' then nullif(p->>'subcategory_id', '')::uuid else subcategory_id end,
      batch_no       = coalesce(v_batch, batch_no),
      hsn_code       = case when p ? 'hsn_code' then nullif(trim(p->>'hsn_code'), '') else hsn_code end,
      price          = coalesce((p->>'price')::numeric, price),
      low_stock_threshold = coalesce((p->>'low_stock_threshold')::int, low_stock_threshold),
      is_active      = coalesce((p->>'is_active')::boolean, is_active)
    where id = v_id returning * into v_prod;
    if not found then raise exception 'Product not found'; end if;
  end if;

  if p ? 'images' and jsonb_typeof(p->'images') = 'array' then
    select array_agg(x->>'url') into v_urls from jsonb_array_elements(p->'images') x;
    with d as (
      delete from public.product_images
       where product_id = v_prod.id and (v_urls is null or url <> all (v_urls))
      returning storage_path)
    select coalesce(jsonb_agg(storage_path) filter (where storage_path is not null), '[]'::jsonb)
      into v_removed from d;

    update public.product_images set is_primary = false where product_id = v_prod.id;
    select coalesce(
             (select ord::int from jsonb_array_elements(p->'images') with ordinality t(x, ord)
               where coalesce((x->>'is_primary')::boolean, false) limit 1), 1) into v_prim;

    for img in select x, ord from jsonb_array_elements(p->'images') with ordinality t(x, ord) loop
      insert into public.product_images(product_id, url, storage_path, sort_order, is_primary)
      values (v_prod.id, img.x->>'url', img.x->>'storage_path',
              coalesce((img.x->>'sort_order')::int, img.ord::int), img.ord = v_prim)
      on conflict (product_id, url) do update
        set sort_order = excluded.sort_order, is_primary = excluded.is_primary,
            storage_path = coalesce(excluded.storage_path, public.product_images.storage_path);
    end loop;
  end if;

  return jsonb_build_object('product', to_jsonb(v_prod), 'removed_storage_paths', v_removed);
end $$;

create or replace function public.admin_delete_product(p_id uuid)
returns jsonb language plpgsql security definer set search_path = public as $$
declare v_paths jsonb;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  select coalesce(jsonb_agg(storage_path) filter (where storage_path is not null), '[]'::jsonb)
    into v_paths from public.product_images where product_id = p_id;
  delete from public.products where id = p_id;
  if not found then raise exception 'Product not found'; end if;
  return jsonb_build_object('deleted', true, 'removed_storage_paths', v_paths);
end $$;

create or replace function public.admin_adjust_stock(p_product_id uuid, p_delta int)
returns public.products language plpgsql security definer set search_path = public as $$
declare v public.products;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  perform set_config('app.stock_reason', 'manual_adjustment', true);
  update public.products set stock = stock + p_delta where id = p_product_id returning * into v;
  perform set_config('app.stock_reason', '', true);
  if v.id is null then raise exception 'Product not found'; end if;   -- NOT FOUND is unreliable after PERFORM
  return v;
end $$;

create or replace function public.admin_list_products(
  p_search text default null, p_filter text default 'all', p_category_id uuid default null,
  p_page int default 1, p_page_size int default 10)
returns jsonb language plpgsql stable security definer set search_path = public as $$
declare v_total bigint; v_rows jsonb;
        v_size int := least(greatest(p_page_size, 1), 100);
        v_off int := greatest(p_page - 1, 0) * least(greatest(p_page_size, 1), 100);
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  select count(*) into v_total from public.v_products_admin p
   where (coalesce(p_search, '') = '' or (p.name || ' ' || p.sku) ilike '%' || public.escape_like(p_search) || '%')
     and (p_filter <> 'low_stock' or p.is_low_stock)
     and (p_category_id is null or p.category_id = p_category_id or p.subcategory_id = p_category_id);
  select coalesce(jsonb_agg(to_jsonb(t)), '[]'::jsonb) into v_rows from (
    select p.id, p.name, p.category_id, p.category_name, p.subcategory_id, p.subcategory_name,
           p.sku, p.batch_no, p.hsn_code, p.price, p.stock, p.low_stock_threshold, p.is_low_stock, p.is_active,
           p.primary_image, p.created_at
      from public.v_products_admin p
     where (coalesce(p_search, '') = '' or (p.name || ' ' || p.sku) ilike '%' || public.escape_like(p_search) || '%')
       and (p_filter <> 'low_stock' or p.is_low_stock)
       and (p_category_id is null or p.category_id = p_category_id or p.subcategory_id = p_category_id)
     order by p.created_at desc offset v_off limit v_size) t;
  return jsonb_build_object('total', v_total, 'page', p_page, 'page_size', v_size, 'offset', v_off, 'rows', v_rows);
end $$;

create or replace function public.admin_get_product(p_id uuid)
returns jsonb language plpgsql stable security definer set search_path = public as $$
declare v jsonb;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  select to_jsonb(p) || jsonb_build_object(
           'images', coalesce((select jsonb_agg(to_jsonb(i) order by i.is_primary desc, i.sort_order)
                                 from public.product_images i where i.product_id = p.id), '[]'::jsonb))
    into v from public.v_products_admin p where p.id = p_id;
  if v is null then raise exception 'Product not found'; end if;
  return v;
end $$;

create or replace function public.admin_save_campaign(p jsonb)
returns jsonb language plpgsql security definer set search_path = public as $$
declare v_id uuid := nullif(p->>'id', '')::uuid; v_c public.festive_campaigns; v_coupon uuid; pid text;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  v_coupon := nullif(p->>'coupon_id', '')::uuid;
  if v_coupon is null and coalesce(p->>'coupon_code', '') <> '' then
    select id into v_coupon from public.coupons where upper(code) = upper(trim(p->>'coupon_code'));
    if v_coupon is null then raise exception 'Coupon code not found'; end if;
  end if;

  if v_id is null then
    insert into public.festive_campaigns(name, banner_image_url, banner_path, discount_percent, coupon_id,
                                         display_order, starts_at, ends_at, is_active)
    values (trim(p->>'name'), p->>'banner_image_url', p->>'banner_path', (p->>'discount_percent')::numeric,
            v_coupon, coalesce((p->>'display_order')::int, 0), nullif(p->>'starts_at', '')::timestamptz,
            nullif(p->>'ends_at', '')::timestamptz, coalesce((p->>'is_active')::boolean, true))
    returning * into v_c;
  else
    update public.festive_campaigns set
      name = coalesce(nullif(trim(p->>'name'), ''), name),
      banner_image_url = coalesce(p->>'banner_image_url', banner_image_url),
      banner_path = case when p ? 'banner_path' then p->>'banner_path' else banner_path end,
      discount_percent = coalesce((p->>'discount_percent')::numeric, discount_percent),
      coupon_id = case when p ? 'coupon_id' or p ? 'coupon_code' then v_coupon else coupon_id end,
      display_order = coalesce((p->>'display_order')::int, display_order),
      starts_at = case when p ? 'starts_at' then nullif(p->>'starts_at', '')::timestamptz else starts_at end,
      ends_at = case when p ? 'ends_at' then nullif(p->>'ends_at', '')::timestamptz else ends_at end,
      is_active = coalesce((p->>'is_active')::boolean, is_active)
    where id = v_id returning * into v_c;
    if not found then raise exception 'Campaign not found'; end if;
  end if;

  if p ? 'product_ids' then
    delete from public.festive_campaign_products where campaign_id = v_c.id;
    for pid in select jsonb_array_elements_text(p->'product_ids') loop
      insert into public.festive_campaign_products(campaign_id, product_id) values (v_c.id, pid::uuid)
      on conflict do nothing;
    end loop;
  end if;

  return to_jsonb(v_c) || jsonb_build_object('product_ids',
    coalesce((select jsonb_agg(product_id) from public.festive_campaign_products where campaign_id = v_c.id), '[]'::jsonb));
end $$;

    from public.festive_campaigns fc

   where fcp.product_id = p_product and fc.is_active

     and (fc.ends_at   is null or fc.ends_at   >= now());
$$;

-- Public safe campaign feed: display fields, a coupon CODE if currently usable, and ids of
-- products that are active AND whose categories are visible.
create or replace function public.get_active_campaigns()
returns table (id uuid, name text, banner_image_url text, discount_percent numeric, display_order int,
               starts_at timestamptz, ends_at timestamptz, coupon_code text, product_ids jsonb)

  select fc.id, fc.name, fc.banner_image_url, fc.discount_percent, fc.display_order, fc.starts_at, fc.ends_at,
         (select c.code from public.coupons c
           where c.id = fc.coupon_id and c.is_active
             and (c.starts_at is null or c.starts_at <= now())
             and (c.expires_at is null or c.expires_at >= now())
             and (c.usage_limit is null or c.used_count < c.usage_limit)),
         coalesce((select jsonb_agg(fcp.product_id)
                     from public.festive_campaign_products fcp
                     join public.products p on p.id = fcp.product_id
                    where fcp.campaign_id = fc.id and p.is_active
                      and public.category_visible(p.category_id)
                      and (p.subcategory_id is null or public.category_visible(p.subcategory_id))), '[]'::jsonb)
    from public.festive_campaigns fc
   where fc.is_active
     and (fc.starts_at is null or fc.starts_at <= now())
     and (fc.ends_at   is null or fc.ends_at   >= now())

$$;

-- pass ALL ids of the list in the new order; validated and applied atomically
create or replace function public.admin_reorder(p_table text, p_ids uuid[])
returns void language plpgsql security definer set search_path = public as $$
declare v_cnt bigint;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  if p_table not in ('hero_banners', 'announcements', 'festive_campaigns') then raise exception 'Table not reorderable'; end if;
  if p_ids is null or array_length(p_ids, 1) is null then raise exception 'Nothing to reorder'; end if;
  if (select count(distinct u) from unnest(p_ids) u) <> array_length(p_ids, 1) then raise exception 'Duplicate ids in payload'; end if;
  execute format('select count(*) from public.%I where id = any ($1)', p_table) into v_cnt using p_ids;
  if v_cnt <> array_length(p_ids, 1) then raise exception 'Unknown id in payload'; end if;
  execute format('update public.%I t set display_order = o.ord from unnest($1) with ordinality as o(id, ord) where t.id = o.id', p_table)
    using p_ids;
end $$;

-- service_role only (called by the public gateway Edge Function after CAPTCHA passes).
create or replace function public.submit_lead(
  p_name text, p_email text, p_phone text, p_message text, p_ip_hash text, p_user_agent text default null)
returns jsonb language plpgsql security definer set search_path = public as $$
declare v_id uuid;
begin
  if coalesce(p_ip_hash, '') = '' then raise exception 'ip_hash required'; end if;
  if length(coalesce(p_message, '')) > 5000 then return jsonb_build_object('ok', false, 'error', 'message_too_long'); end if;

end $$;

-- Dates are store local calendar days (store_settings.timezone), inclusive on both ends.
create or replace function public.admin_list_leads(
  p_search text default null, p_status public.lead_status default null,
  p_from date default null, p_to date default null,
  p_sort text default 'newest', p_page int default 1, p_page_size int default 10)
returns jsonb language plpgsql stable security definer set search_path = public as $$
declare v_total bigint; v_rows jsonb;
        v_size int := least(greatest(p_page_size, 1), 100);
        v_off int := greatest(p_page - 1, 0) * least(greatest(p_page_size, 1), 100);
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  select count(*) into v_total from public.leads l
   where (p_status is null or l.status = p_status)
     and (p_from is null or l.received_at >= public.day_start(p_from))
     and (p_to   is null or l.received_at <  public.day_start(p_to + 1))
     and (coalesce(p_search, '') = '' or (l.name || ' ' || l.email || ' ' || coalesce(l.phone, ''))
          ilike '%' || public.escape_like(p_search) || '%');
  select coalesce(jsonb_agg(to_jsonb(t)), '[]'::jsonb) into v_rows from (
    select l.id, l.name, l.email, l.phone, l.message, l.status, l.received_at, l.responded_at
      from public.leads l
     where (p_status is null or l.status = p_status)
       and (p_from is null or l.received_at >= public.day_start(p_from))
       and (p_to   is null or l.received_at <  public.day_start(p_to + 1))
       and (coalesce(p_search, '') = '' or (l.name || ' ' || l.email || ' ' || coalesce(l.phone, ''))
            ilike '%' || public.escape_like(p_search) || '%')
     order by case when p_sort = 'oldest' then l.received_at end asc, l.received_at desc
     offset v_off limit v_size) t;
  return jsonb_build_object('total', v_total, 'page', p_page, 'page_size', v_size, 'offset', v_off, 'rows', v_rows);
end $$;

-- STRICT STATE MACHINE (enforced in the DB for every writer, service_role included)
--   pending -> in_progress | cancelled ;  in_progress -> delivered | cancelled
--   delivered and cancelled are terminal (returns/refunds are separate workflows)
create or replace function public.orders_status_guard()
returns trigger language plpgsql as $$
begin
  if new.status is distinct from old.status then
    if not ((old.status = 'pending'     and new.status in ('in_progress', 'cancelled'))
         or (old.status = 'in_progress' and new.status in ('delivered', 'cancelled'))) then
      raise exception 'Invalid order status change: % -> % (delivered and cancelled are final)', old.status, new.status;
    end if;
  end if;
  return new;
end $$;
create trigger trg_orders_status_guard before update of status on public.orders
  for each row execute function public.orders_status_guard();

-- keep orders.payment_status derived from the ledger (never trust the client)
create or replace function public.recompute_payment_status(p_order uuid)
returns void language plpgsql security definer set search_path = public as $$
declare v_paid numeric; v_ref numeric; v_failed boolean; v_pending boolean; v_ostatus public.order_status;
        v_status public.payment_status;
begin
  select status into v_ostatus from public.orders where id = p_order;
  if not found then return; end if;
  select coalesce(sum(amount) filter (where kind = 'charge' and status = 'succeeded'), 0),
         coalesce(sum(amount) filter (where kind = 'refund' and status = 'succeeded'), 0),
         coalesce(bool_or(kind = 'charge' and status = 'failed'), false),
         coalesce(bool_or(kind = 'charge' and status = 'pending'), false)
    into v_paid, v_ref, v_failed, v_pending
    from public.payments where order_id = p_order;
  v_status := case
    when v_paid > 0 and v_ref >= v_paid then 'refunded'
    when v_ref > 0 then 'partially_refunded'
    when v_paid > 0 then 'paid'
    when v_ostatus = 'cancelled' then 'voided'
    when v_failed and not v_pending then 'failed'
    else 'pending' end;
  update public.orders set payment_status = v_status where id = p_order and payment_status is distinct from v_status;
end $$;

create or replace function public.payments_sync_order()
returns trigger language plpgsql security definer set search_path = public as $$

  return null;
end $$;
create trigger trg_payments_sync after insert or update or delete on public.payments

-- Only delivery metadata and the issued -> cancelled flag may change. Everything else is frozen.
create or replace function public.invoices_immutable()
returns trigger language plpgsql as $$
begin
  if tg_op = 'DELETE' then raise exception 'Invoices cannot be deleted'; end if;
  if (to_jsonb(new) - array['pdf_path', 'emailed_at', 'status', 'cancelled_at'])
     is distinct from (to_jsonb(old) - array['pdf_path', 'emailed_at', 'status', 'cancelled_at']) then
    raise exception 'Invoices are immutable';
  end if;
  if new.status <> old.status and not (old.status = 'issued' and new.status = 'cancelled') then
    raise exception 'Invalid invoice status change';
  end if;
  return new;
end $$;
create trigger trg_invoices_immutable before update or delete on public.invoices
  for each row execute function public.invoices_immutable();

  order_id        uuid references public.orders(id) on delete cascade,
  invoice_id      uuid references public.invoices(id) on delete restrict,
  recipient       text not null,
  status          text not null default 'pending' check (status in ('pending', 'sending', 'sent', 'failed')),
  attempts        int not null default 0,
  last_error      text,
  next_attempt_at timestamptz not null default now(),
  created_at      timestamptz not null default now(),
  sent_at         timestamptz,
  unique (kind, invoice_id)
);
create index notification_outbox_due_idx on public.notification_outbox(next_attempt_at)
  where status in ('pending', 'sending', 'failed');

-- Issue the invoice for a confirmed order. Idempotent: calling twice returns the same invoice.
-- Internal: callers are place_order (COD), admin_update_order (COD accept) and apply_payment_event (online).
create or replace function public.issue_invoice(p_order_id uuid)
returns public.invoices language plpgsql security definer set search_path = public as $$
declare
  o public.orders; s public.store_settings; inv public.invoices; v_pay record;
  v_today date; v_fy text; v_n int; v_no text; v_paid numeric; v_state text; v_break jsonb;
begin
  select * into inv from public.invoices where order_id = p_order_id;
  if found then return inv; end if;

  select * into o from public.orders where id = p_order_id for update;
  if not found then raise exception 'Order not found'; end if;
  select * into inv from public.invoices where order_id = p_order_id;     -- issued while we waited for the lock
  if found then return inv; end if;
  if o.status = 'cancelled' then raise exception 'Cannot invoice a cancelled order'; end if;

  select * into s from public.store_settings limit 1;
  v_today := public.store_today();
  v_fy := public.fy_label(v_today);
  insert into public.invoice_counters(fy, last_no) values (v_fy, 1)
  on conflict (fy) do update set last_no = public.invoice_counters.last_no + 1
  returning last_no into v_n;
  v_no := s.invoice_prefix || '/' || v_fy || '/' || lpad(v_n::text, 6, '0');

  select coalesce(sum(amount) filter (where kind = 'charge' and status = 'succeeded'), 0)
    into v_paid from public.payments where order_id = o.id;
  select provider, provider_order_id, provider_payment_id, method, captured_at into v_pay
    from public.payments
   where order_id = o.id and kind = 'charge' and status = 'succeeded' order by created_at limit 1;

  v_state := o.shipping_address->>'state';
  v_break := case
    when o.tax_total = 0 then jsonb_build_object('type', 'none')
    when coalesce(trim(s.seller_state), '') = '' then
      jsonb_build_object('type', 'single', 'rate', o.tax_rate_percent, 'tax', o.tax_total)
    when lower(trim(s.seller_state)) = lower(trim(v_state)) then
      jsonb_build_object('type', 'cgst_sgst', 'rate_each', o.tax_rate_percent / 2,
                         'cgst', round(o.tax_total / 2, 2), 'sgst', o.tax_total - round(o.tax_total / 2, 2))
    else jsonb_build_object('type', 'igst', 'rate', o.tax_rate_percent, 'igst', o.tax_total) end;

  insert into public.invoices(order_id, invoice_number, seller, buyer, place_of_supply, currency,
      subtotal, discount_total, shipping_fee, taxable_value, tax_total, tax_rate_percent, prices_include_tax,
      tax_breakdown, total, payment_method, payment_status, amount_paid, paid_at,
      payment_provider, provider_order_id, provider_payment_id, payment_instrument)
  values (o.id, v_no,
      jsonb_build_object('legal_name', s.seller_legal_name, 'store_name', s.store_name, 'gstin', s.seller_gstin,
                         'address', s.seller_address, 'state', s.seller_state, 'email', s.seller_email, 'phone', s.seller_phone),
      jsonb_build_object('name', o.customer_name, 'email', o.customer_email, 'phone', o.customer_phone,
                         'shipping_address', o.shipping_address),
      v_state, s.currency,
      o.subtotal, o.discount_total, o.shipping_fee, o.total - o.tax_total, o.tax_total, o.tax_rate_percent,
      o.prices_include_tax, v_break, o.total, o.payment_method,
      case when v_paid > 0 then 'paid' else 'pending' end, v_paid, v_pay.captured_at,
      v_pay.provider, v_pay.provider_order_id, v_pay.provider_payment_id, v_pay.method)
  returning * into inv;

  insert into public.invoice_items(invoice_id, line_no, product_name, sku, hsn_code, quantity, list_price, unit_price, line_total)
  select inv.id, row_number() over (order by oi.product_name, oi.id), oi.product_name, oi.sku, oi.hsn_code,
         oi.quantity, oi.list_price, oi.unit_price, oi.line_total
    from public.order_items oi where oi.order_id = o.id;

  update public.orders set confirmed_at = coalesce(confirmed_at, now()) where id = o.id;
  insert into public.order_status_history(order_id, status, is_read, note)
  values (o.id, o.status, o.is_read, 'Invoice ' || v_no || ' issued');
  insert into public.notification_outbox(kind, order_id, invoice_id, recipient)
  values ('invoice_email', o.id, inv.id, o.customer_email) on conflict do nothing;
  return inv;
end $$;

-- Customer reads their own invoice (header + lines + payment references)
create or replace function public.get_my_invoice(p_order_id uuid)
returns jsonb language plpgsql stable security definer set search_path = public as $$
declare v jsonb;
begin
  if auth.uid() is null then raise exception 'Authentication required'; end if;
  select jsonb_build_object('invoice', to_jsonb(i),
           'items', coalesce((select jsonb_agg(to_jsonb(it) order by it.line_no)
                                from public.invoice_items it where it.invoice_id = i.id), '[]'::jsonb))
    into v
    from public.invoices i join public.orders o on o.id = i.order_id
   where o.id = p_order_id and o.user_id = auth.uid();
  if v is null then raise exception 'Invoice not available yet'; end if;
  return v;
end $$;

create or replace function public.admin_list_invoices(
  p_search text default null, p_page int default 1, p_page_size int default 10)
returns jsonb language plpgsql stable security definer set search_path = public as $$
declare v_total bigint; v_rows jsonb;
        v_size int := least(greatest(p_page_size, 1), 100);
        v_off int := greatest(p_page - 1, 0) * least(greatest(p_page_size, 1), 100);
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  select count(*) into v_total from public.invoices i
   where coalesce(p_search, '') = ''
      or concat_ws(' ', i.invoice_number, i.buyer->>'name', i.buyer->>'email', i.provider_payment_id)
         ilike '%' || public.escape_like(p_search) || '%';
  select coalesce(jsonb_agg(to_jsonb(t)), '[]'::jsonb) into v_rows from (
    select i.id, i.order_id, i.invoice_number, i.status, i.issued_at, i.total, i.payment_method,
           i.payment_status, i.provider_payment_id, i.emailed_at, i.buyer->>'name' as customer_name
      from public.invoices i
     where coalesce(p_search, '') = ''
        or concat_ws(' ', i.invoice_number, i.buyer->>'name', i.buyer->>'email', i.provider_payment_id)
           ilike '%' || public.escape_like(p_search) || '%'
     order by i.issued_at desc offset v_off limit v_size) t;
  return jsonb_build_object('total', v_total, 'page', p_page, 'page_size', v_size, 'offset', v_off, 'rows', v_rows);
end $$;

-- Outbox for the send-invoices Edge Function (service_role). Safe for parallel workers.
create or replace function public.claim_outbox(p_limit int default 10)
returns setof public.notification_outbox language sql security definer set search_path = public as $$
  with c as (
    select id from public.notification_outbox
     where status in ('pending', 'failed', 'sending') and attempts < 8 and next_attempt_at <= now()
     order by id limit least(greatest(p_limit, 1), 50)
     for update skip locked)
  update public.notification_outbox o
     set attempts = o.attempts + 1, status = 'sending', next_attempt_at = now() + interval '10 minutes'
    from c where o.id = c.id
  returning o.*;
$$;

create or replace function public.mark_outbox(p_id bigint, p_ok boolean, p_error text default null)
returns void language plpgsql security definer set search_path = public as $$
declare v public.notification_outbox;
begin
  if p_ok then
    update public.notification_outbox set status = 'sent', sent_at = now(), last_error = null
     where id = p_id returning * into v;
    if v.invoice_id is not null then
      update public.invoices set emailed_at = now() where id = v.invoice_id;
    end if;
  else
    update public.notification_outbox
       set status = 'failed', last_error = left(p_error, 500),
           next_attempt_at = now() + (attempts * interval '15 minutes')
     where id = p_id;
  end if;
end $$;

-- Orders that hold money that must go back (cancelled but paid, late payment, double payment)
create or replace view public.v_orders_needing_refund with (security_invoker = true) as
select o.id, o.order_number, o.status, o.total, p.paid, p.refunded, p.refund_pending,
       (p.paid - p.refunded - p.refund_pending) as refundable
  from public.orders o
  join lateral (
    select coalesce(sum(amount) filter (where kind = 'charge' and status = 'succeeded'), 0) as paid,
           coalesce(sum(amount) filter (where kind = 'refund' and status = 'succeeded'), 0) as refunded,
           coalesce(sum(amount) filter (where kind = 'refund' and status = 'pending'), 0) as refund_pending
      from public.payments where order_id = o.id) p on true
 where public.is_admin()
   and p.paid - p.refunded - p.refund_pending > 0
   and (o.status = 'cancelled' or p.paid - p.refunded > o.total + 0.009);

create or replace view public.v_payment_events_failed with (security_invoker = true) as
select id, provider, event_id, event_type, status, error, attempts, received_at, processed_at
  from public.payment_webhook_events
 where status = 'failed' and public.is_admin();

-- internal: create a refund row (validates against captured money, counts pending refunds)
create or replace function public._create_refund_row(
  p_order uuid, p_amount numeric, p_reason text, p_ref text, p_actor uuid, p_status text default 'succeeded')
returns public.payments language plpgsql security definer set search_path = public as $$
declare v_paid numeric; v_ref numeric; v_pend numeric; v_provider text; v_cur char(3); v_row public.payments;
begin
  if p_amount is null or p_amount <= 0 then raise exception 'Refund amount must be positive'; end if;
  if p_status not in ('pending', 'succeeded') then raise exception 'Invalid refund status'; end if;
  perform 1 from public.orders where id = p_order for update;
  if not found then raise exception 'Order not found'; end if;
  select coalesce(sum(amount) filter (where kind = 'charge' and status = 'succeeded'), 0),
         coalesce(sum(amount) filter (where kind = 'refund' and status = 'succeeded'), 0),
         coalesce(sum(amount) filter (where kind = 'refund' and status = 'pending'), 0)
    into v_paid, v_ref, v_pend from public.payments where order_id = p_order;
  if v_paid = 0 then raise exception 'No captured payment to refund'; end if;
  if v_ref + v_pend + p_amount > v_paid + 0.009 then
    raise exception 'Refund exceeds captured amount (captured %, refunded %, in progress %)', v_paid, v_ref, v_pend;
  end if;
  select provider, currency into v_provider, v_cur from public.payments
   where order_id = p_order and kind = 'charge' and status = 'succeeded' order by created_at limit 1;
  insert into public.payments(order_id, provider, kind, provider_payment_id, amount, currency, status, reason, created_by)
  values (p_order, v_provider, 'refund',
          case when p_status = 'succeeded' then coalesce(p_ref, 'manual-' || gen_random_uuid()::text) else p_ref end,
          p_amount, v_cur, p_status, p_reason, p_actor)
  returning * into v_row;
  return v_row;
end $$;

-- Admin records a refund that was ALREADY issued outside the system (COD cash refund, bank transfer,
-- or a Razorpay refund made in their dashboard). For Razorpay always pass the refund id (rfnd_xxx)
-- so the webhook does not double count it.
create or replace function public.admin_record_refund(
  p_order_id uuid, p_amount numeric, p_reason text default null, p_provider_refund_id text default null)
returns public.payments language plpgsql security definer set search_path = public as $$
declare v_provider text;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  select provider into v_provider from public.payments
   where order_id = p_order_id and kind = 'charge' and status = 'succeeded' order by created_at limit 1;
  if v_provider = 'razorpay' and coalesce(p_provider_refund_id, '') = '' then
    raise exception 'Pass the Razorpay refund id, or issue the refund through the refund endpoint';
  end if;
  return public._create_refund_row(p_order_id, p_amount, p_reason, p_provider_refund_id, auth.uid(), 'succeeded');
end $$;

-- Gateway refunds (service_role, called by the refund Edge Function after it verified the admin JWT).
create or replace function public.begin_gateway_refund(
  p_order_id uuid, p_amount numeric, p_reason text, p_actor uuid, p_payment_id text default null)
returns jsonb language plpgsql security definer set search_path = public as $$
declare v_pay public.payments; v_row public.payments;
begin
  if not exists (select 1 from public.profiles where id = p_actor and role = 'admin' and is_active) then
    raise exception 'Actor is not an active admin';
  end if;
  select * into v_pay from public.payments
   where order_id = p_order_id and kind = 'charge' and status = 'succeeded' and provider = 'razorpay'
     and (p_payment_id is null or provider_payment_id = p_payment_id)
   order by created_at limit 1;
  if not found then raise exception 'No captured Razorpay payment found for this order'; end if;
  if p_amount > v_pay.amount + 0.009 then raise exception 'Refund exceeds this payment amount'; end if;
  v_row := public._create_refund_row(p_order_id, p_amount, p_reason, null, p_actor, 'pending');
  return jsonb_build_object('refund_row_id', v_row.id, 'razorpay_payment_id', v_pay.provider_payment_id,
                            'amount_minor', round(p_amount * 100)::bigint, 'currency', v_pay.currency);
end $$;

create or replace function public.finish_gateway_refund(
  p_row_id uuid, p_provider_refund_id text, p_error text default null)
returns jsonb language plpgsql security definer set search_path = public as $$
declare v_provider text;
begin
  if p_error is not null then
    update public.payments set status = 'failed', reason = left(coalesce(reason, '') || ' | ' || p_error, 500)
     where id = p_row_id and kind = 'refund' and status = 'pending';
    return jsonb_build_object('status', 'failed');
  end if;
  select provider into v_provider from public.payments where id = p_row_id;
  if exists (select 1 from public.payments where provider = v_provider and provider_payment_id = p_provider_refund_id) then
    delete from public.payments where id = p_row_id and status = 'pending';    -- webhook won the race
    return jsonb_build_object('status', 'already_recorded');
  end if;
  update public.payments set provider_payment_id = p_provider_refund_id
   where id = p_row_id and kind = 'refund' and status = 'pending' and provider_payment_id is null;
  return jsonb_build_object('status', 'pending_webhook');
end $$;

-- internal: restock + undo coupon + void unpaid charges + cancel the invoice (every cancellation path)
create or replace function public._reverse_order_effects(p_order uuid)
returns void language plpgsql security definer set search_path = public as $$
declare r record;
begin
  perform set_config('app.stock_reason', 'order_cancelled', true);
  for r in select product_id, quantity from public.order_items
            where order_id = p_order and product_id is not null order by product_id loop
    update public.products set stock = stock + r.quantity where id = r.product_id;
  end loop;
  perform set_config('app.stock_reason', '', true);
  with d as (delete from public.coupon_redemptions where order_id = p_order returning coupon_id)
  update public.coupons c set used_count = greatest(c.used_count - 1, 0)
   where c.id in (select coupon_id from d);
  update public.payments set status = 'voided', reason = coalesce(reason, 'Order cancelled')
   where order_id = p_order and kind = 'charge' and status = 'pending';
  update public.invoices set status = 'cancelled', cancelled_at = now()
   where order_id = p_order and status = 'issued';
end $$;

-- internal: amount checked coupon logic
create or replace function public._coupon_discount(c public.coupons, p_user uuid, p_amount numeric)
returns numeric language plpgsql stable security definer set search_path = public as $$
declare d numeric;
begin
  if not c.is_active or (c.starts_at is not null and c.starts_at > now())
     or (c.expires_at is not null and c.expires_at < now()) then
    raise exception 'Invalid or expired coupon';
  end if;
  if c.usage_limit is not null and c.used_count >= c.usage_limit then raise exception 'Coupon usage limit reached'; end if;
  if (select count(*) from public.coupon_redemptions where coupon_id = c.id and user_id = p_user) >= c.per_user_limit then
    raise exception 'Coupon already used';
  end if;
  if p_amount < c.min_order_amount then raise exception 'Minimum order amount not met for this coupon'; end if;
  d := case when c.discount_type = 'percent' then round(p_amount * c.discount_value / 100, 2) else c.discount_value end;
  if c.max_discount is not null then d := least(d, c.max_discount); end if;
  return least(d, p_amount);
end $$;

-- Checkout (atomic + idempotent + throttled).
-- p_items: [{"product_id":"..","quantity":2}]
-- p_shipping: {"full_name","phone","line1","line2","city","state","postal_code","country"}
-- p_idempotency_key: client generated UUID per checkout attempt; repeats return the SAME order.
-- Tax: if store_settings.prices_include_tax, tax is extracted from the price (price stays the price);
-- otherwise it is added on top. Shipping is inside the taxable base.
create or replace function public.place_order(
  p_items jsonb, p_shipping jsonb, p_payment_method text default 'cod',
  p_coupon_code text default null, p_notes text default null, p_idempotency_key text default null)
returns public.orders language plpgsql security definer set search_path = public as $$
declare
  v_uid uuid := auth.uid(); v_profile public.profiles; v_settings public.store_settings; v_order public.orders;
  r record; v_prod public.products; v_pct numeric; v_unit numeric;
  v_subtotal numeric := 0; v_discount numeric := 0; v_cdisc numeric := 0; v_ship numeric := 0; v_tax numeric := 0;
  v_net numeric; v_base numeric; v_total numeric; v_open int;
  v_coupon public.coupons; v_first text; v_count int := 0; v_name text; v_lines jsonb := '[]'::jsonb;
  k text; v_ship_addr jsonb; v_pids uuid[] := '{}';
begin
  if v_uid is null then raise exception 'Authentication required'; end if;

  if p_idempotency_key is not null then
    if length(p_idempotency_key) > 100 then raise exception 'Idempotency key too long'; end if;
    perform pg_advisory_xact_lock(hashtextextended(v_uid::text || ':' || p_idempotency_key, 0));
    select * into v_order from public.orders where user_id = v_uid and idempotency_key = p_idempotency_key;
    if found then return v_order; end if;
  end if;

  if p_items is null or jsonb_typeof(p_items) <> 'array' or jsonb_array_length(p_items) = 0 then
    raise exception 'Cart is empty'; end if;
  if p_shipping is null or jsonb_typeof(p_shipping) <> 'object' then raise exception 'Shipping address required'; end if;
  foreach k in array array['full_name', 'phone', 'line1', 'city', 'state', 'postal_code'] loop
    if coalesce(trim(p_shipping->>k), '') = '' then raise exception 'Shipping address field missing: %', k; end if;
  end loop;
  if p_payment_method not in ('cod', 'online') then raise exception 'Unsupported payment method'; end if;

  select * into v_profile from public.profiles where id = v_uid and is_active;
  if not found then raise exception 'Account inactive'; end if;
  select * into v_settings from public.store_settings limit 1;

  -- throttles (a rejected call rolls back its own counter, so only real orders are counted)
  if not public.rate_limit_hit('order:' || v_uid::text, 10, 3600) then
    raise exception 'Too many orders, please try again later';
  end if;
  select count(*) into v_open from public.orders where user_id = v_uid and status = 'pending';
  if v_open >= v_settings.max_open_orders then
    raise exception 'You have too many open orders. Complete or cancel one first.';
  end if;

  v_ship_addr := jsonb_build_object(
    'full_name', trim(p_shipping->>'full_name'), 'phone', trim(p_shipping->>'phone'),
    'line1', trim(p_shipping->>'line1'), 'line2', nullif(trim(p_shipping->>'line2'), ''),
    'city', trim(p_shipping->>'city'), 'state', trim(p_shipping->>'state'),
    'postal_code', trim(p_shipping->>'postal_code'), 'country', coalesce(nullif(trim(p_shipping->>'country'), ''), 'IN'));

  for r in select (value->>'product_id')::uuid as pid, sum((value->>'quantity')::int)::int as qty
             from jsonb_array_elements(p_items) group by 1 order by 1 loop
    if r.qty is null or r.qty <= 0 or r.qty > 100 then raise exception 'Invalid quantity'; end if;
    select * into v_prod from public.products where id = r.pid and is_active for update;
    if not found or not public.category_visible(v_prod.category_id)
       or (v_prod.subcategory_id is not null and not public.category_visible(v_prod.subcategory_id)) then
      raise exception 'A product in your cart is no longer available';
    end if;
    if v_prod.stock < r.qty then raise exception 'Insufficient stock for %', v_prod.name; end if;

    v_pct  := public.active_discount_percent(v_prod.id);
    v_unit := round(v_prod.price * (1 - v_pct / 100), 2);
    v_subtotal := v_subtotal + v_prod.price * r.qty;
    v_discount := v_discount + (v_prod.price - v_unit) * r.qty;
    v_count := v_count + 1;
    v_pids := v_pids || v_prod.id;
    if v_first is null then v_first := v_prod.name; end if;
    v_lines := v_lines || jsonb_build_object('product_id', v_prod.id, 'name', v_prod.name, 'sku', v_prod.sku,
        'hsn', v_prod.hsn_code, 'qty', r.qty, 'unit', v_unit, 'list', v_prod.price,
        'image', (select url from public.product_images where product_id = v_prod.id
                   order by is_primary desc, sort_order limit 1));
  end loop;

  if coalesce(trim(p_coupon_code), '') <> '' then
    if not v_settings.coupons_stack_with_campaigns and v_discount > 0 then
      raise exception 'Coupons cannot be combined with sale prices';
    end if;
    select * into v_coupon from public.coupons where upper(code) = upper(trim(p_coupon_code)) for update;
    if not found then raise exception 'Invalid or expired coupon'; end if;
    begin
      v_cdisc := public._coupon_discount(v_coupon, v_uid, v_subtotal - v_discount);
    exception when others then
      raise exception 'Invalid or expired coupon';       -- one generic message: no coupon oracle
    end;
  end if;

  v_net  := v_subtotal - v_discount - v_cdisc;
  v_ship := case when v_settings.free_shipping_threshold is not null and v_net >= v_settings.free_shipping_threshold
                 then 0 else v_settings.shipping_flat_fee end;
  v_base := v_net + v_ship;
  if v_settings.prices_include_tax then
    v_tax := round(v_base * v_settings.tax_rate_percent / (100 + v_settings.tax_rate_percent), 2);
    v_total := v_base;
  else
    v_tax := round(v_base * v_settings.tax_rate_percent / 100, 2);
    v_total := v_base + v_tax;
  end if;
  if v_total <= 0 then raise exception 'Order total must be greater than zero'; end if;
  v_name := v_first || case when v_count > 1 then ' +' || (v_count - 1) || ' more' else '' end;

  insert into public.orders(order_name, user_id, idempotency_key, customer_name, customer_email, customer_phone,
      shipping_address, subtotal, discount_total, coupon_discount, shipping_fee, tax_total, tax_rate_percent,
      prices_include_tax, total, coupon_id, coupon_code, payment_method, notes)
  values (v_name, v_uid, p_idempotency_key, v_ship_addr->>'full_name', v_profile.email, v_ship_addr->>'phone',
      v_ship_addr, v_subtotal, v_discount + v_cdisc, v_cdisc, v_ship, v_tax, v_settings.tax_rate_percent,
      v_settings.prices_include_tax, v_total, v_coupon.id, v_coupon.code, p_payment_method, left(p_notes, 1000))
  returning * into v_order;

  perform set_config('app.stock_reason', 'order_placed', true);
  for r in select value as x from jsonb_array_elements(v_lines) loop
    insert into public.order_items(order_id, product_id, product_name, sku, hsn_code, image_url,
        unit_price, list_price, quantity, line_total)
    values (v_order.id, (r.x->>'product_id')::uuid, r.x->>'name', r.x->>'sku', r.x->>'hsn', r.x->>'image',
        (r.x->>'unit')::numeric, (r.x->>'list')::numeric, (r.x->>'qty')::int,
        (r.x->>'unit')::numeric * (r.x->>'qty')::int);
    update public.products set stock = stock - (r.x->>'qty')::int where id = (r.x->>'product_id')::uuid;
  end loop;
  perform set_config('app.stock_reason', '', true);

  if v_coupon.id is not null then
    update public.coupons set used_count = used_count + 1 where id = v_coupon.id;
    insert into public.coupon_redemptions(coupon_id, user_id, order_id, amount)
    values (v_coupon.id, v_uid, v_order.id, v_cdisc);
  end if;

  -- open payment intent: becomes 'succeeded' ONLY via verified Razorpay event (online) or delivery (COD)
  insert into public.payments(order_id, provider, kind, amount, currency, status)
  values (v_order.id, case when p_payment_method = 'cod' then 'cod' else 'razorpay' end, 'charge',
          v_order.total, v_settings.currency, 'pending');
  insert into public.order_status_history(order_id, status, is_read, note, changed_by)
  values (v_order.id, 'pending', false, 'Order placed', v_uid);

  delete from public.cart_items where user_id = v_uid and product_id = any (v_pids);   -- only what was bought

  if p_payment_method = 'cod' and v_settings.cod_invoice_on = 'placed' then
    perform public.issue_invoice(v_order.id);
  end if;

  select * into v_order from public.orders where id = v_order.id;
  return v_order;
end $$;

-- Volatile on purpose: it writes to rate_limits. Every failure returns the same generic message.
create or replace function public.validate_coupon(p_code text, p_subtotal numeric)
returns jsonb language plpgsql security definer set search_path = public as $$
declare c public.coupons; d numeric;
begin
  if auth.uid() is null or not public.is_active_user() then raise exception 'Authentication required'; end if;
  if not public.rate_limit_hit('coupon:' || auth.uid()::text, 20, 3600) then
    return jsonb_build_object('valid', false, 'message', 'Too many attempts, please try again later');
  end if;
  select * into c from public.coupons where upper(code) = upper(trim(p_code));
  if not found then return jsonb_build_object('valid', false, 'message', 'Invalid or expired coupon'); end if;
  begin
    d := public._coupon_discount(c, auth.uid(), p_subtotal);
    return jsonb_build_object('valid', true, 'code', c.code, 'discount', d);
  exception when others then
    return jsonb_build_object('valid', false, 'message', 'Invalid or expired coupon');
  end;
end $$;

-- Customer cancels own order (only while pending)
create or replace function public.cancel_my_order(p_order_id uuid, p_reason text default null)
returns public.orders language plpgsql security definer set search_path = public as $$
declare o public.orders;
begin
  if not public.is_active_user() then raise exception 'Account inactive'; end if;
  select * into o from public.orders where id = p_order_id and user_id = auth.uid() for update;
  if not found then raise exception 'Order not found'; end if;
  if o.status <> 'pending' then raise exception 'Only pending orders can be cancelled'; end if;
  perform public._reverse_order_effects(o.id);
  update public.orders set status = 'cancelled', cancelled_at = now(), is_read = false, read_at = null, read_by = null
   where id = o.id;
  select * into o from public.orders where id = o.id;
  insert into public.order_status_history(order_id, status, is_read, note, changed_by)
  values (o.id, 'cancelled', false, coalesce(p_reason, 'Cancelled by customer'), auth.uid());
  return o;
end $$;

-- ---------------------------------------------------------------------
-- 15c. RAZORPAY ONLINE PAYMENT FUNCTIONS
-- ---------------------------------------------------------------------
-- Step 2a: customer (user JWT, via Edge fn) asks to pay. Returns what the Edge fn needs to create
-- (or reuse) the Razorpay order. Never trusts an amount from the client.
create or replace function public.begin_online_payment(p_order_id uuid)
returns jsonb language plpgsql security definer set search_path = public as $$
declare o public.orders; v_win int; v_cur char(3); v_pay public.payments;
begin
  if auth.uid() is null or not public.is_active_user() then raise exception 'Authentication required'; end if;
  if not public.rate_limit_hit('pay:' || auth.uid()::text, 30, 3600) then
    return jsonb_build_object('ok', false, 'error', 'rate_limited');
  end if;
  select * into o from public.orders where id = p_order_id and user_id = auth.uid() for update;
  if not found then raise exception 'Order not found'; end if;
  if o.payment_method <> 'online' then raise exception 'Not an online payment order'; end if;
  if o.status <> 'pending' then raise exception 'Order is no longer payable'; end if;
  if o.payment_status in ('paid', 'partially_refunded', 'refunded') then raise exception 'Order is already paid'; end if;
  select payment_window_minutes, currency into v_win, v_cur from public.store_settings limit 1;
  if o.created_at < now() - make_interval(mins => v_win) then raise exception 'Payment window has expired'; end if;
  select * into v_pay from public.payments
   where order_id = o.id and kind = 'charge' and status = 'pending' and provider_payment_id is null
   order by created_at limit 1;
  if not found then raise exception 'No payable charge found for this order'; end if;
  return jsonb_build_object('ok', true, 'order_id', o.id, 'order_number', o.order_number,
    'amount_minor', round(o.total * 100)::bigint, 'currency', v_cur, 'payment_id', v_pay.id,
    'razorpay_order_id', v_pay.provider_order_id,
    'expires_at', o.created_at + make_interval(mins => v_win),
    'customer', jsonb_build_object('name', o.customer_name, 'email', o.customer_email, 'phone', o.customer_phone));
end $$;

-- Step 2b (service_role): store the Razorpay order id on the open intent. Idempotent.
create or replace function public.attach_razorpay_order(p_order_id uuid, p_rzp_order_id text, p_amount_minor bigint)
returns jsonb language plpgsql security definer set search_path = public as $$
declare o public.orders; v_pay public.payments;
begin
  if coalesce(p_rzp_order_id, '') = '' then raise exception 'Razorpay order id required'; end if;
  select * into o from public.orders where id = p_order_id for update;
  if not found then raise exception 'Order not found'; end if;
  if o.status <> 'pending' then raise exception 'Order is not payable'; end if;
  if p_amount_minor is distinct from round(o.total * 100)::bigint then raise exception 'Amount mismatch'; end if;
  if exists (select 1 from public.payments where provider = 'razorpay' and provider_order_id = p_rzp_order_id
                and order_id <> p_order_id) then
    raise exception 'Razorpay order already attached to another order';
  end if;
  select * into v_pay from public.payments
   where order_id = o.id and kind = 'charge' and status = 'pending' and provider_payment_id is null
   order by created_at limit 1 for update;
  if not found then raise exception 'No open payment intent for this order'; end if;
  if v_pay.provider_order_id is not null then
    return jsonb_build_object('razorpay_order_id', v_pay.provider_order_id, 'reused', true);
  end if;
  update public.payments set provider = 'razorpay', provider_order_id = p_rzp_order_id where id = v_pay.id;
  return jsonb_build_object('razorpay_order_id', p_rzp_order_id, 'reused', false);
end $$;

-- Verified Razorpay events -> ledger. service_role ONLY (the Edge fn verifies the signature first).
-- Idempotent per (provider, event_id). A previously FAILED event can be replayed.
-- Amounts are in minor units (paise). Supported events:
--   payment.captured | order.paid  -> charge succeeded, order confirmed, invoice issued
--   payment.failed                 -> failed attempt row (the order stays payable until the window ends)
--   refund.processed | refund.failed
create or replace function public.apply_payment_event(
  p_provider text, p_event_id text, p_event_type text,
  p_provider_order_id text, p_provider_payment_id text,
  p_amount_minor bigint, p_currency text, p_raw jsonb,
  p_method text default null, p_error_code text default null, p_error_description text default null,
  p_parent_payment_id text default null, p_order_id uuid default null)
returns jsonb language plpgsql security definer set search_path = public as $$
declare
  v_evt uuid; v_order public.orders; v_settings public.store_settings; v_oid uuid := p_order_id;
  v_paid numeric; v_ref numeric; v_row uuid; v_amount numeric; v_inv public.invoices;
  v_late boolean := false;
begin
  if coalesce(p_provider, '') = '' or coalesce(p_event_id, '') = '' then
    raise exception 'provider and event_id are required';
  end if;

  insert into public.payment_webhook_events(provider, event_id, event_type, payload)
  values (p_provider, p_event_id, p_event_type, p_raw)
  on conflict (provider, event_id) do update
     set received_at = now(), status = 'received', error = null, processed_at = null,
         attempts = public.payment_webhook_events.attempts + 1, payload = excluded.payload
   where public.payment_webhook_events.status = 'failed'
  returning id into v_evt;
  if v_evt is null then return jsonb_build_object('status', 'duplicate'); end if;

  if p_event_type not in ('payment.captured', 'order.paid', 'payment.failed', 'refund.processed', 'refund.failed') then
    update public.payment_webhook_events set status = 'ignored', processed_at = now() where id = v_evt;
    return jsonb_build_object('status', 'ignored');
  end if;

  begin
    select * into v_settings from public.store_settings limit 1;

    if v_oid is null and p_provider_order_id is not null then
      select order_id into v_oid from public.payments
       where provider = p_provider and provider_order_id = p_provider_order_id order by created_at limit 1;
    end if;
    if v_oid is null and p_parent_payment_id is not null then
      select order_id into v_oid from public.payments
       where provider = p_provider and provider_payment_id = p_parent_payment_id limit 1;
    end if;
    if v_oid is null then raise exception 'Order could not be resolved'; end if;

    select * into v_order from public.orders where id = v_oid for update;
    if not found then raise exception 'Order not found'; end if;

    if p_provider_order_id is not null and exists (
         select 1 from public.payments
          where provider = p_provider and provider_order_id = p_provider_order_id and order_id <> v_oid) then
      raise exception 'Gateway order belongs to a different order';
    end if;
    if p_event_type in ('payment.captured', 'order.paid', 'payment.failed') then
      if p_provider_order_id is null or not exists (
           select 1 from public.payments
            where order_id = v_oid and provider = p_provider and provider_order_id = p_provider_order_id) then
        raise exception 'Gateway order is not attached to this order';
      end if;
      if coalesce(p_provider_payment_id, '') = '' then raise exception 'Payment id missing'; end if;
    end if;

    if p_event_type in ('payment.captured', 'order.paid') then
      if upper(coalesce(p_currency, '')) <> upper(v_settings.currency) then raise exception 'Currency mismatch'; end if;
      if p_amount_minor is null or p_amount_minor <> round(v_order.total * 100)::bigint then
        raise exception 'Amount mismatch (paid %, expected %)', p_amount_minor, round(v_order.total * 100)::bigint;
      end if;

      if not exists (select 1 from public.payments where provider = p_provider
                        and provider_payment_id = p_provider_payment_id and kind = 'charge' and status = 'succeeded') then
        update public.payments
           set status = 'succeeded', provider_payment_id = p_provider_payment_id, method = p_method,
               captured_at = now(), raw = p_raw, error_code = null, error_description = null
         where id = (select id from public.payments
                      where order_id = v_oid and provider = p_provider and kind = 'charge' and status = 'pending'
                        and provider_payment_id is null and provider_order_id = p_provider_order_id
                      order by created_at limit 1)
        returning id into v_row;
        if v_row is null then          -- intent already voided/closed, or a second payment: keep it in the ledger
          insert into public.payments(order_id, provider, provider_order_id, kind, provider_payment_id,
                                      amount, currency, status, method, captured_at, raw)
          values (v_oid, p_provider, p_provider_order_id, 'charge', p_provider_payment_id,
                  v_order.total, v_settings.currency, 'succeeded', p_method, now(), p_raw)
          on conflict (provider, provider_payment_id) where provider_payment_id is not null do nothing;
        end if;
      end if;

      select coalesce(sum(amount) filter (where kind = 'charge' and status = 'succeeded'), 0)
        into v_paid from public.payments where order_id = v_oid;

      if v_order.status = 'cancelled' then
        v_late := true;                 -- money arrived after cancellation or expiry: refund needed, no invoice
      else
        update public.orders set confirmed_at = coalesce(confirmed_at, now()) where id = v_oid;
        insert into public.order_status_history(order_id, status, is_read, note)
        values (v_oid, v_order.status, v_order.is_read, 'Payment received');
        if v_paid >= v_order.total - 0.009 then v_inv := public.issue_invoice(v_oid); end if;
      end if;

    elsif p_event_type = 'payment.failed' then
      insert into public.payments(order_id, provider, provider_order_id, kind, provider_payment_id, amount, currency,
                                  status, method, error_code, error_description, raw)
      values (v_oid, p_provider, p_provider_order_id, 'charge', p_provider_payment_id, v_order.total, v_settings.currency,
              'failed', p_method, p_error_code, left(p_error_description, 500), p_raw)
      on conflict (provider, provider_payment_id) where provider_payment_id is not null do nothing;

    else   -- refund.processed | refund.failed
      if coalesce(p_provider_payment_id, '') = '' then raise exception 'Refund id missing'; end if;
      if p_event_type = 'refund.failed' then
        update public.payments set status = 'failed', raw = p_raw
         where kind = 'refund' and provider = p_provider and provider_payment_id = p_provider_payment_id
           and status = 'pending';
      else
        if p_amount_minor is null or p_amount_minor <= 0 then raise exception 'Refund amount missing'; end if;
        v_amount := p_amount_minor / 100.0;
        update public.payments set status = 'succeeded', raw = p_raw
         where kind = 'refund' and provider = p_provider and provider_payment_id = p_provider_payment_id
           and status = 'pending'
        returning id into v_row;
        if v_row is null and not exists (select 1 from public.payments
                                          where provider = p_provider and provider_payment_id = p_provider_payment_id) then
          -- refund created outside this system (for example in the Razorpay dashboard)
          select coalesce(sum(amount) filter (where kind = 'charge' and status = 'succeeded'), 0),
                 coalesce(sum(amount) filter (where kind = 'refund' and status = 'succeeded'), 0)
            into v_paid, v_ref from public.payments where order_id = v_oid;
          if v_paid = 0 then raise exception 'No captured payment to refund'; end if;
          if v_ref + v_amount > v_paid + 0.009 then raise exception 'Refund exceeds captured amount'; end if;
          insert into public.payments(order_id, provider, kind, provider_payment_id, amount, currency, status, raw)
          values (v_oid, p_provider, 'refund', p_provider_payment_id, v_amount, v_settings.currency, 'succeeded', p_raw);
        end if;
      end if;
    end if;

    update public.payment_webhook_events
       set status = 'processed', processed_at = now(),
           error = case when v_late then 'Paid after cancellation: refund required' end
     where id = v_evt;
    return jsonb_build_object('status', 'processed', 'order_id', v_oid, 'needs_refund', v_late,
                              'invoice_number', v_inv.invoice_number);
  exception when others then
    -- keep the event row (marked failed, replayable) and tell the caller; never a retry storm
    update public.payment_webhook_events
       set status = 'failed', error = left(sqlerrm, 500), processed_at = now() where id = v_evt;
    return jsonb_build_object('status', 'failed', 'error', sqlerrm);
  end;
end $$;

-- Cron: cancel unpaid online orders after payment_window_minutes (+5 minutes grace for payments in
-- flight) and release their stock. Optionally cancel COD orders nobody opened (cod_auto_cancel_hours).
--   select cron.schedule('expire-unpaid-orders', '*/5 * * * *', 'select public.expire_unpaid_orders()');
create or replace function public.expire_unpaid_orders()
returns int language plpgsql security definer set search_path = public as $$
declare r record; n int := 0; v_mins int; v_cod int;
begin
  select payment_window_minutes, cod_auto_cancel_hours into v_mins, v_cod from public.store_settings limit 1;
  for r in select id from public.orders
            where status = 'pending'
              and ((payment_method = 'online' and payment_status in ('pending', 'failed')
                    and created_at < now() - make_interval(mins => v_mins + 5))
                or (payment_method = 'cod' and v_cod is not null and is_read = false
                    and created_at < now() - make_interval(hours => v_cod)))
            for update skip locked loop
    perform public._reverse_order_effects(r.id);
    update public.orders set status = 'cancelled', cancelled_at = now(), is_read = true, read_at = now() where id = r.id;
    insert into public.order_status_history(order_id, status, is_read, note)
    values (r.id, 'cancelled', true, 'Cancelled automatically (payment window or confirmation expired)');
    n := n + 1;
  end loop;
  return n;
end $$;

-- Orders table: search + (status, read/unread, dates) combinable + sort + pagination.
-- Dates are store local calendar days, inclusive.
create or replace function public.admin_list_orders(
  p_search text default null, p_status public.order_status default null, p_is_read boolean default null,
  p_from date default null, p_to date default null,
  p_sort text default 'newest', p_page int default 1, p_page_size int default 10)
returns jsonb language plpgsql stable security definer set search_path = public as $$
declare v_total bigint; v_rows jsonb;
        v_size int := least(greatest(p_page_size, 1), 100);
        v_off int := greatest(p_page - 1, 0) * least(greatest(p_page_size, 1), 100);
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  select count(*) into v_total from public.orders o
   where (p_status is null or o.status = p_status)
     and (p_is_read is null or o.is_read = p_is_read)
     and (p_from is null or o.created_at >= public.day_start(p_from))
     and (p_to   is null or o.created_at <  public.day_start(p_to + 1))
     and (coalesce(p_search, '') = '' or
          (o.order_number || ' ' || o.order_name || ' ' || o.customer_name || ' ' || o.customer_email || ' ' || coalesce(o.customer_phone, ''))
          ilike '%' || public.escape_like(p_search) || '%');
  select coalesce(jsonb_agg(to_jsonb(t)), '[]'::jsonb) into v_rows from (
    select o.id, o.order_number, o.order_name, o.customer_name, o.customer_email, o.customer_phone,
           o.total, o.status, o.payment_method, o.payment_status, o.is_read, o.confirmed_at, o.created_at
      from public.orders o
     where (p_status is null or o.status = p_status)
       and (p_is_read is null or o.is_read = p_is_read)
       and (p_from is null or o.created_at >= public.day_start(p_from))
       and (p_to   is null or o.created_at <  public.day_start(p_to + 1))
       and (coalesce(p_search, '') = '' or
            (o.order_number || ' ' || o.order_name || ' ' || o.customer_name || ' ' || o.customer_email || ' ' || coalesce(o.customer_phone, ''))
            ilike '%' || public.escape_like(p_search) || '%')
     order by case when p_sort = 'oldest' then o.created_at end asc, o.created_at desc
     offset v_off limit v_size) t;
  return jsonb_build_object('total', v_total, 'page', p_page, 'page_size', v_size, 'offset', v_off, 'rows', v_rows);
end $$;

create or replace function public.admin_get_order(p_id uuid)
returns jsonb language plpgsql stable security definer set search_path = public as $$
declare v jsonb;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  select jsonb_build_object(
    'order',    to_jsonb(o),
    'items',    coalesce((select jsonb_agg(to_jsonb(i)) from public.order_items i where i.order_id = o.id), '[]'::jsonb),
    'history',  coalesce((select jsonb_agg(to_jsonb(h) order by h.created_at) from public.order_status_history h where h.order_id = o.id), '[]'::jsonb),
    'payments', coalesce((select jsonb_agg(to_jsonb(p) - 'raw' order by p.created_at) from public.payments p where p.order_id = o.id), '[]'::jsonb),
    'returns',  coalesce((select jsonb_agg(to_jsonb(rr)) from public.return_requests rr where rr.order_id = o.id), '[]'::jsonb),
    'invoice',  (select to_jsonb(inv) - 'seller' - 'buyer' from public.invoices inv where inv.order_id = o.id))
   into v from public.orders o where o.id = p_id;
  if v is null then raise exception 'Order not found'; end if;
  return v;
end $$;

-- Admin: read/unread + strict status transitions
create or replace function public.admin_update_order(
  p_order_id uuid, p_is_read boolean default null,
  p_status public.order_status default null, p_note text default null)
returns public.orders language plpgsql security definer set search_path = public as $$
declare o public.orders; v_read boolean; v_status public.order_status; v_upd int; v_settings public.store_settings;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  select * into o from public.orders where id = p_order_id for update;
  if not found then raise exception 'Order not found'; end if;
  select * into v_settings from public.store_settings limit 1;

  v_status := coalesce(p_status, o.status);
  v_read   := coalesce(p_is_read, o.is_read);

  if v_status <> o.status then
    if not ((o.status = 'pending'     and v_status in ('in_progress', 'cancelled'))
         or (o.status = 'in_progress' and v_status in ('delivered', 'cancelled'))) then
      raise exception 'Invalid order status change: % -> % (delivered and cancelled are final)', o.status, v_status;
    end if;
    if v_status in ('in_progress', 'delivered') then
      v_read := true;
      if o.payment_method = 'online' and o.payment_status <> 'paid' then
        raise exception 'Online order has not been paid yet';
      end if;
    end if;
    if v_status = 'cancelled' then perform public._reverse_order_effects(o.id); end if;
  end if;

  update public.orders set
      is_read      = v_read,
      read_at      = case when v_read and not o.is_read then now() when not v_read then null else o.read_at end,
      read_by      = case when v_read and not o.is_read then auth.uid() when not v_read then null else o.read_by end,
      status       = v_status,
      delivered_at = case when v_status = 'delivered' and o.status <> 'delivered' then now() else o.delivered_at end,
      cancelled_at = case when v_status = 'cancelled' and o.status <> 'cancelled' then now() else o.cancelled_at end
   where id = o.id;

  -- COD cash is collected on delivery: this is the moment it becomes revenue
  if v_status = 'delivered' and o.status <> 'delivered' and o.payment_method = 'cod' then
    update public.payments set status = 'succeeded', captured_at = now(), method = 'cod',
           provider_payment_id = coalesce(provider_payment_id, 'cod-' || o.order_number), created_by = auth.uid()
     where order_id = o.id and provider = 'cod' and kind = 'charge' and status = 'pending';
    get diagnostics v_upd = row_count;
    if v_upd = 0 then
      insert into public.payments(order_id, provider, kind, provider_payment_id, amount, currency, status, method, captured_at, created_by)
      values (o.id, 'cod', 'charge', 'cod-' || o.order_number || '-' || substr(gen_random_uuid()::text, 1, 6),
              o.total, v_settings.currency, 'succeeded', 'cod', now(), auth.uid());
    end if;
  end if;

  -- COD order is confirmed when the admin accepts it (unless the store confirms COD at placement)
  if v_status = 'in_progress' and o.status = 'pending' and o.payment_method = 'cod'
     and v_settings.cod_invoice_on = 'accepted' then
    perform public.issue_invoice(o.id);
  end if;

  select * into o from public.orders where id = o.id;
  insert into public.order_status_history(order_id, status, is_read, note, changed_by)
  values (o.id, o.status, o.is_read, p_note, auth.uid());
  return o;
end $$;

-- Returns workflow --------------------------------------------------
-- p_items: [{"order_item_id":"..","quantity":1}]
create or replace function public.request_return(p_order_id uuid, p_items jsonb, p_reason text)
returns public.return_requests language plpgsql security definer set search_path = public as $$
declare o public.orders; v_ret public.return_requests; x jsonb; oi public.order_items; v_taken int; v_days int; v_qty int;
begin
  if not public.is_active_user() then raise exception 'Account inactive'; end if;
  select * into o from public.orders where id = p_order_id and user_id = auth.uid() for update;
  if not found then raise exception 'Order not found'; end if;
  if o.status <> 'delivered' then raise exception 'Only delivered orders can be returned'; end if;
  select return_window_days into v_days from public.store_settings limit 1;
  if o.delivered_at < now() - make_interval(days => v_days) then raise exception 'Return window has closed'; end if;
  if coalesce(trim(p_reason), '') = '' then raise exception 'Reason required'; end if;
  if p_items is null or jsonb_typeof(p_items) <> 'array' or jsonb_array_length(p_items) = 0 then
    raise exception 'Select at least one item'; end if;

  insert into public.return_requests(order_id, user_id, reason) values (o.id, auth.uid(), trim(p_reason))
  returning * into v_ret;
  for x in select * from jsonb_array_elements(p_items) loop
    select * into oi from public.order_items where id = (x->>'order_item_id')::uuid and order_id = o.id;
    if not found then raise exception 'Item does not belong to this order'; end if;
    v_qty := (x->>'quantity')::int;
    select coalesce(sum(ri.quantity), 0) into v_taken
      from public.return_request_items ri join public.return_requests rr on rr.id = ri.return_id
     where ri.order_item_id = oi.id and rr.status <> 'rejected';
    if v_qty is null or v_qty <= 0 or v_qty + v_taken > oi.quantity then raise exception 'Invalid return quantity'; end if;
    insert into public.return_request_items(return_id, order_item_id, quantity) values (v_ret.id, oi.id, v_qty);
  end loop;
  return v_ret;
end $$;

create or replace function public.admin_review_return(p_id uuid, p_approve boolean, p_note text default null)
returns public.return_requests language plpgsql security definer set search_path = public as $$
declare v public.return_requests;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  update public.return_requests
     set status = case when p_approve then 'approved'::public.return_status else 'rejected'::public.return_status end,
         admin_note = p_note, reviewed_by = auth.uid()
   where id = p_id and status = 'requested' returning * into v;
  if not found then raise exception 'Return not found or already reviewed'; end if;
  return v;
end $$;

-- Goods received: optionally restock, optionally refund. The refund is capped at the value of the
-- returned items. Razorpay orders: start the refund through the refund endpoint FIRST
-- (begin_gateway_refund), then complete the return. COD orders: the refund is recorded here.
create or replace function public.admin_complete_return(
  p_id uuid, p_restock boolean default true, p_refund_amount numeric default 0, p_provider_refund_id text default null)
returns public.return_requests language plpgsql security definer set search_path = public as $$
declare v public.return_requests; r record; v_cap numeric; v_provider text; v_started numeric;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  select * into v from public.return_requests where id = p_id for update;
  if not found or v.status <> 'approved' then raise exception 'Return must be approved first'; end if;

  select coalesce(sum(oi.unit_price * ri.quantity), 0) into v_cap
    from public.return_request_items ri join public.order_items oi on oi.id = ri.order_item_id
   where ri.return_id = v.id;
  if coalesce(p_refund_amount, 0) > v_cap + 0.009 then
    raise exception 'Refund (%) exceeds the value of the returned items (%)', p_refund_amount, v_cap;
  end if;

  if p_restock then
    perform set_config('app.stock_reason', 'return_restock', true);
    for r in select oi.product_id, ri.quantity from public.return_request_items ri
               join public.order_items oi on oi.id = ri.order_item_id
              where ri.return_id = v.id and oi.product_id is not null order by oi.product_id loop
      update public.products set stock = stock + r.quantity where id = r.product_id;
    end loop;
    perform set_config('app.stock_reason', '', true);
  end if;

  if coalesce(p_refund_amount, 0) > 0 then
    select provider into v_provider from public.payments
     where order_id = v.order_id and kind = 'charge' and status = 'succeeded' order by created_at limit 1;
    if v_provider = 'razorpay' then
      select coalesce(sum(amount), 0) into v_started from public.payments
       where order_id = v.order_id and kind = 'refund' and status in ('pending', 'succeeded');
      if v_started + 0.009 < p_refund_amount then
        raise exception 'Start the Razorpay refund through the refund endpoint first, then complete the return';
      end if;
    else
      perform public._create_refund_row(v.order_id, p_refund_amount, 'Return ' || v.id, p_provider_refund_id, auth.uid(), 'succeeded');
    end if;
  end if;

  update public.return_requests set status = 'completed', restocked = p_restock,
         refund_amount = coalesce(p_refund_amount, 0), completed_at = now(), reviewed_by = coalesce(reviewed_by, auth.uid())
   where id = v.id returning * into v;
  return v;
end $$;

-- ---------------------------------------------------------------------
-- 16. ANALYTICS (page views / unique visitors / sessions, aggregated)
--   Intake ONLY via the public gateway Edge Function (service_role):
--   consent    -> first party random id (visitor_key 'c:<uuid>')

--   Days roll over at store local midnight (store_settings.timezone).
-- ---------------------------------------------------------------------
create table public.page_views (
  id          bigint generated always as identity primary key,

  session_id  text not null,

  user_agent  text,
  consented   boolean not null default false,
  created_at  timestamptz not null default now()
);
create index page_views_created_idx on public.page_views(created_at desc);

create table public.site_visitors (
  visitor_key text primary key,
  consented   boolean not null default false,
  first_seen  timestamptz not null default now(),
  last_seen   timestamptz not null default now(),
  page_views  bigint not null default 1
);
create index site_visitors_last_seen_idx on public.site_visitors(last_seen);
create table public.site_sessions (
  session_id  text primary key,
  visitor_key text not null,
  started_at  timestamptz not null default now(),
  last_seen   timestamptz not null default now(),
  page_views  int not null default 1
);
create index site_sessions_last_seen_idx on public.site_sessions(last_seen);

  primary key (day, visitor_key)

  day             date primary key,
  page_views      bigint not null default 0,
  unique_visitors bigint not null default 0,
  sessions        bigint not null default 0

create or replace function public.track_page_view(
  p_visitor_key text, p_session_id text, p_path text, p_referrer text, p_user_agent text,
  p_ip_hash text, p_consented boolean default false)
returns jsonb language plpgsql security definer set search_path = public as $$
declare v_new_visitor boolean; v_new_session boolean; v_new_today int; v_day date := public.store_today();
begin
  if coalesce(p_visitor_key, '') = '' or coalesce(p_session_id, '') = '' or coalesce(p_ip_hash, '') = '' then
    raise exception 'visitor_key, session_id and ip_hash are required';
  end if;
  if left(coalesce(p_path, ''), 1) <> '/' then raise exception 'Invalid path'; end if;
  if not public.rate_limit_hit('pv:ip:' || p_ip_hash, 300, 60)
     or not public.rate_limit_hit('pv:v:' || p_visitor_key, 120, 60) then
    return jsonb_build_object('ok', false, 'error', 'rate_limited');
  end if;

  insert into public.page_views(visitor_key, session_id, path, referrer, user_agent, consented)
  values (p_visitor_key, p_session_id, left(p_path, 500), left(p_referrer, 500), left(p_user_agent, 300), coalesce(p_consented, false));

  insert into public.site_visitors(visitor_key, consented) values (p_visitor_key, coalesce(p_consented, false))
  on conflict (visitor_key) do update set last_seen = now(), page_views = public.site_visitors.page_views + 1
  returning (xmax = 0) into v_new_visitor;

  insert into public.site_sessions(session_id, visitor_key) values (p_session_id, p_visitor_key)
  on conflict (session_id) do update set last_seen = now(), page_views = public.site_sessions.page_views + 1
  returning (xmax = 0) into v_new_session;

  insert into public.site_visitor_days(day, visitor_key) values (v_day, p_visitor_key) on conflict do nothing;
  get diagnostics v_new_today = row_count;

  insert into public.analytics_daily(day, page_views, unique_visitors, sessions)
  values (v_day, 1, v_new_today, case when v_new_session then 1 else 0 end)
  on conflict (day) do update set page_views = public.analytics_daily.page_views + 1,
         unique_visitors = public.analytics_daily.unique_visitors + excluded.unique_visitors,
         sessions = public.analytics_daily.sessions + excluded.sessions;
  return jsonb_build_object('ok', true);
end $$;

-- retention: raw events and visitor keys expire, daily aggregates stay.
--   select cron.schedule('purge-analytics', '15 3 * * *', 'select public.purge_old_analytics(90)');
create or replace function public.purge_old_analytics(p_keep_days int default 90)
returns void language plpgsql security definer set search_path = public as $$
begin
  delete from public.page_views where created_at < now() - make_interval(days => p_keep_days);
  delete from public.site_visitor_days where day < public.store_today() - p_keep_days;
  delete from public.site_sessions where last_seen < now() - make_interval(days => p_keep_days);
  delete from public.site_visitors where last_seen < now() - make_interval(days => p_keep_days);
  perform public.purge_rate_limits();
end $$;

-- retention for operational logs. Orders, payments, invoices and order history are NEVER purged
-- (keep tax records for the legally required period). Failed webhook events are kept for follow up.
--   select cron.schedule('purge-logs', '30 3 * * 0', 'select public.purge_old_logs()');
create or replace function public.purge_old_logs(
  p_webhook_days int default 180, p_audit_days int default 400, p_outbox_days int default 90)
returns void language plpgsql security definer set search_path = public as $$
begin
  delete from public.payment_webhook_events
   where status in ('processed', 'ignored') and received_at < now() - make_interval(days => p_webhook_days);
  delete from public.audit_logs where created_at < now() - make_interval(days => p_audit_days);
  delete from public.notification_outbox where status = 'sent' and sent_at < now() - make_interval(days => p_outbox_days);
end $$;

create or replace function public.admin_analytics(
  p_from date default public.store_today() - 29, p_to date default public.store_today())
returns jsonb language plpgsql stable security definer set search_path = public as $$
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  if p_to < p_from or p_to - p_from > 366 then raise exception 'Invalid date range (max 366 days)'; end if;
  return jsonb_build_object(
    'from', p_from, 'to', p_to,
    'totals', (select jsonb_build_object('page_views', coalesce(sum(page_views), 0),
                      'unique_visitors', coalesce(sum(unique_visitors), 0), 'sessions', coalesce(sum(sessions), 0))
                 from public.analytics_daily where day between p_from and p_to),
    'daily', coalesce((select jsonb_agg(to_jsonb(d) order by d.day)
                         from public.analytics_daily d where d.day between p_from and p_to), '[]'::jsonb));
end $$;

-- ---------------------------------------------------------------------
-- 17. ADMIN KPI FUNCTIONS
-- ---------------------------------------------------------------------
-- Revenue = money actually collected (succeeded charges) minus succeeded refunds.
-- Pending / failed / voided payments and unpaid orders never count.
-- total_visitors = sum of daily unique visitors (visitor days). Visitors without consent get a daily
-- rotating key, so a lifetime unique count is not possible without a persistent identifier.
create or replace function public.admin_dashboard_stats()
returns jsonb language plpgsql stable security definer set search_path = public as $$
declare v_gross numeric; v_ref numeric;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  select coalesce(sum(amount) filter (where kind = 'charge' and status = 'succeeded'), 0),
         coalesce(sum(amount) filter (where kind = 'refund' and status = 'succeeded'), 0)
    into v_gross, v_ref from public.payments;
  return jsonb_build_object(
    'total_revenue',    v_gross - v_ref,
    'gross_collected',  v_gross,
    'refunded',         v_ref,
    'refunds_pending',  (select count(*) from public.v_orders_needing_refund),
    'failed_payment_events', (select count(*) from public.payment_webhook_events where status = 'failed'),
    'total_orders',     (select count(*) from public.orders),
    'total_categories', (select count(*) from public.categories where parent_id is null),
    'total_visitors',   (select coalesce(sum(unique_visitors), 0) from public.analytics_daily),
    'page_views',       (select coalesce(sum(page_views), 0) from public.analytics_daily),
    'sessions',         (select coalesce(sum(sessions), 0) from public.analytics_daily));
end $$;

create or replace function public.admin_unread_orders(p_page int default 1, p_page_size int default 10)
returns jsonb language plpgsql stable security definer set search_path = public as $$
declare v_size int := least(greatest(p_page_size, 1), 100);
        v_off int := greatest(p_page - 1, 0) * least(greatest(p_page_size, 1), 100);
        v_total bigint; v_rows jsonb;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  select count(*) into v_total from public.orders where is_read = false;
  select coalesce(jsonb_agg(to_jsonb(t)), '[]'::jsonb) into v_rows from (
    select id, order_number, order_name, customer_name, customer_email, customer_phone, total, status,
           payment_method, payment_status, created_at
      from public.orders where is_read = false order by created_at desc offset v_off limit v_size) t;
  return jsonb_build_object('total', v_total, 'page', p_page, 'page_size', v_size, 'offset', v_off, 'rows', v_rows);
end $$;

       d.pct as discount_percent,

create or replace function public.audit_row()
returns trigger language plpgsql security definer set search_path = public as $$
declare v_pii text[] := array['shipping_address','customer_name','customer_email','customer_phone',
                              'email','phone','full_name','name','message','ip_hash','user_agent','avatar_url','notes',
                              'raw','payload','error_description'];
begin
  insert into public.audit_logs(actor_id, table_name, record_id, action, old_data, new_data)
  values (auth.uid(), tg_table_name,
          coalesce(to_jsonb(new)->>'id', to_jsonb(old)->>'id', to_jsonb(new)->>'policy_type', to_jsonb(old)->>'policy_type'),
          tg_op,
          case when tg_op in ('UPDATE', 'DELETE') then to_jsonb(old) - v_pii end,
          case when tg_op in ('INSERT', 'UPDATE') then to_jsonb(new) - v_pii end);
  return null;
end $$;

-- Future objects must NOT be exposed by accident: remove the Supabase default grants for new objects.

  begin alter default privileges for role postgres in schema public revoke all on tables from anon, authenticated; exception when others then null; end;
  begin alter default privileges for role postgres in schema public revoke all on sequences from anon, authenticated; exception when others then null; end;
  begin alter default privileges for role postgres in schema public revoke all on functions from anon, authenticated; exception when others then null; end;
  begin alter default privileges for role supabase_admin in schema public revoke all on tables from anon, authenticated; exception when others then null; end;
  begin alter default privileges for role supabase_admin in schema public revoke all on sequences from anon, authenticated; exception when others then null; end;
  begin alter default privileges for role supabase_admin in schema public revoke all on functions from anon, authenticated; exception when others then null; end;
  begin alter default privileges for role postgres revoke execute on functions from public; exception when others then null; end;
end $$;

-- 2. helper: root-first path {category, sub, subsub} ----------------------------
create or replace function public.category_path(p_id uuid)
returns uuid[] language sql stable security definer set search_path = public as $$
  with recursive up as (
    select id, parent_id, 1 as lvl from public.categories where id = p_id
    union all
    select c.id, c.parent_id, up.lvl + 1
      from public.categories c join up on c.id = up.parent_id
     where up.lvl < 6)
  select array_agg(id order by lvl desc) from up;
$$;
revoke execute on function public.category_path(uuid) from public, anon, authenticated;

-- 3. categories: allow 3 levels ----------------------------------------------------
create or replace function public.categories_before_write()
returns trigger language plpgsql as $$
declare v_pp uuid; v_ppp uuid; v_depth int := 0; v_height int := 0;
begin
  if new.slug is null or new.slug = '' then
    new.slug := public.slugify(new.name) || '-' || substr(gen_random_uuid()::text, 1, 6);
  end if;
  if new.parent_id is not null and coalesce(current_setting('app.bulk_reorder', true), '') <> 'on' then
    select parent_id into v_pp from public.categories where id = new.parent_id;
    if not found then raise exception 'Parent category not found'; end if;
    v_depth := 1;
    if v_pp is not null then
      v_depth := 2;
      select parent_id into v_ppp from public.categories where id = v_pp;
      if v_ppp is not null then
        raise exception 'Only three levels supported: category > subcategory > sub subcategory';
      end if;
    end if;
    if tg_op = 'UPDATE' then
      if exists (select 1 from public.categories c join public.categories g on g.parent_id = c.id
                  where c.parent_id = new.id) then v_height := 2;
      elsif exists (select 1 from public.categories where parent_id = new.id) then v_height := 1; end if;
      if v_depth + v_height > 2 then
        raise exception 'This move would exceed three category levels';
      end if;
    end if;
  end if;
  return new;
end $$;

create or replace function public.categories_recalc_visibility()
returns trigger language plpgsql security definer set search_path = public as $$
declare v_prev text := coalesce(current_setting('app.bulk_reorder', true), '');
begin
  perform set_config('app.bulk_reorder', 'on', true);
  with recursive sub(id, vis) as (
    select c.id,
           c.is_active and coalesce((select p.is_visible from public.categories p where p.id = c.parent_id), true)
      from public.categories c where c.id = new.id
    union
    select c.id, c.is_active and s.vis
      from public.categories c join sub s on c.parent_id = s.id)
  update public.categories c set is_visible = sub.vis
    from sub where c.id = sub.id and c.is_visible is distinct from sub.vis;
  perform set_config('app.bulk_reorder', v_prev, true);
  return null;
end $$;

create or replace function public.categories_after_move()
returns trigger language plpgsql security definer set search_path = public as $$
declare r record; v_path uuid[];
begin
  if new.parent_id is distinct from old.parent_id then
    for r in select p.id, coalesce(p.sub_subcategory_id, p.subcategory_id, p.category_id) as leaf
               from public.products p
              where new.id in (p.category_id, p.subcategory_id, p.sub_subcategory_id) loop
      v_path := public.category_path(r.leaf);
      if array_length(v_path, 1) > 3 then raise exception 'Only three levels supported'; end if;
      update public.products
         set category_id = v_path[1], subcategory_id = v_path[2], sub_subcategory_id = v_path[3]
       where id = r.id;
    end loop;
  end if;
  return null;
end $$;

create or replace function public.reorder_categories(p_items jsonb)
returns void language plpgsql security definer set search_path = public as $$
declare v_ids uuid[]; v_n int;
begin
  if not public.is_admin() then raise exception 'Admin only'; end if;
  if p_items is null or jsonb_typeof(p_items) <> 'array' or jsonb_array_length(p_items) = 0 then
    raise exception 'Nothing to reorder';
  end if;
  select array_agg((x->>'id')::uuid) into v_ids from jsonb_array_elements(p_items) x;
  v_n := array_length(v_ids, 1);
  if (select count(distinct u) from unnest(v_ids) u) <> v_n then raise exception 'Duplicate category ids in payload'; end if;
  if (select count(*) from public.categories where id = any (v_ids)) <> v_n then raise exception 'Unknown category id in payload'; end if;
  if exists (select 1 from jsonb_array_elements(p_items) x
              where nullif(x->>'parent_id', '') is not null
                and not exists (select 1 from public.categories c where c.id = (x->>'parent_id')::uuid)) then
    raise exception 'Unknown parent category in payload';
  end if;

  perform set_config('app.bulk_reorder', 'on', true);
  update public.categories c
     set parent_id  = nullif(x->>'parent_id', '')::uuid,
         sort_order = coalesce((x->>'sort_order')::int, c.sort_order)
    from jsonb_array_elements(p_items) x
   where c.id = (x->>'id')::uuid;
  perform set_config('app.bulk_reorder', '', true);

  if exists (select 1 from public.categories c1
               join public.categories c2 on c2.id = c1.parent_id
               join public.categories c3 on c3.id = c2.parent_id
               join public.categories c4 on c4.id = c3.parent_id) then
    raise exception 'Only three levels supported: category > subcategory > sub subcategory';
  end if;
end $$;

-- 4. products: validate the 3 level path ------------------------------------------
create or replace function public.products_before_write()
returns trigger language plpgsql as $$
begin
  if new.slug is null or new.slug = '' then
    new.slug := public.slugify(new.name) || '-' || substr(gen_random_uuid()::text, 1, 6);
  end if;
  if not exists (select 1 from public.categories where id = new.category_id and parent_id is null) then
    raise exception 'Product category must be a top level category';
  end if;
  if new.subcategory_id is not null and not exists (
       select 1 from public.categories where id = new.subcategory_id and parent_id = new.category_id) then
    raise exception 'Subcategory does not belong to the selected category';
  end if;
  if new.sub_subcategory_id is not null then
    if new.subcategory_id is null then raise exception 'Select a subcategory before a sub subcategory'; end if;
    if not exists (select 1 from public.categories
                    where id = new.sub_subcategory_id and parent_id = new.subcategory_id) then
      raise exception 'Sub subcategory does not belong to the selected subcategory';
    end if;
  end if;
  if tg_op = 'UPDATE' then new.sku := old.sku; end if;
  return new;
end $$;
